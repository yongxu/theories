#!/usr/bin/env python3
"""AI-written (the coordinator's session, 23 November 2026). Put a landed clause at the head of the board's Queue
line. Run in a checkout root, after the landing's gate has passed, with the clause in a file of one line:

    python3 -B /Users/ren/tmp/rw/words14/landed_clause.py /path/to/clause.txt

The clause's form: head words with no "(" and no ";" before the word "landed", then
"landed DD Month HH:MM WIB (content tip `…`, on `…`; …; not pushed)." : one line, one parenthesis that closes
at the end, a full stop after it. It edits `_process/BECOMING-WORKBOARD.md` only, commits nothing, and changes no
other clause. Then: `python3 -B tools/research/gate.py board --base main`, commit the board, fast-forward."""
import re
import sys

sys.path.insert(0, 'tools/research')
import gate

B = '_process/BECOMING-WORKBOARD.md'
P = '- **Queue:** '
L = re.compile(r"^[A-Za-z`«][^;(]{0,220}?\blanded\b[^;(]{0,40}\(content tip")
clause = open(sys.argv[1], encoding='utf-8').read().strip()
assert '\n' not in clause, 'the clause must be one line'
assert L.match(clause), 'the clause does not begin "<head words> landed <date> (content tip"'
assert clause.endswith(').'), 'the clause must end with ").": one parenthesis, then a full stop'
assert not gate.HEAD_CLAUSE.search(clause), 'the clause is worded as a head clause'
depth = 0
for i, ch in enumerate(clause):
    depth += ch in '([' and 1 or ch in ')]' and -1 or 0
    assert depth >= 0, 'a bracket closes that was not opened'
    assert not (depth == 0 and ch in '.;' and i + 1 < len(clause)), 'a "." or ";" outside the parenthesis splits it'
assert depth == 0, 'a bracket is left open'
s = open(B, encoding='utf-8').read()
assert s.count('\n' + P) == 1, 'the board must have exactly one Queue line'
before = gate.clauses(gate.queue_line(s))
assert clause not in before, 'the clause is already on the line'
s = s.replace('\n' + P, '\n' + P + clause + ' ')
open(B, 'w', encoding='utf-8').write(s)
after = gate.clauses(gate.queue_line(s))
assert after[0] == clause and after[1:] == before, 'the line did not come out as the clause followed by what stood'
print('clause written at the head of the Queue line;', sum(bool(L.match(x)) for x in after), 'landings on the line;',
      len(clause), 'characters')
