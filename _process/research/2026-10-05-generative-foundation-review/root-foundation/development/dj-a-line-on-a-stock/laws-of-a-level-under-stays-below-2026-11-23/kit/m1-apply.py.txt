import re


def reflow(match):
    words = re.findall(r"(?:[^\s`]|`[^`]*`)+", match.group(1))
    lines, line = [], "/--"
    for word in words + ["-/"]:
        if len(line) + 1 + len(word) > 100:
            lines.append(line)
            line = word
        else:
            line += " " + word
    lines.append(line)
    return "\n".join(lines) + "\n" + match.group(2)
