def decl(text, name):
    m = re.search(r"^(private )?theorem " + re.escape(name) + r"[ \n]", text, flags=re.M)
    if not m: return None
    start = m.start()
    end = text.find("\n\n", start)
    body = text[start:end]
    # statement ends at the first ":=" at end of a line, or (pattern-matching form) before the first "\n  |"
    k = body.find(":=\n")
    if k < 0 or (body.find("\n  | ") >= 0 and body.find("\n  | ") < k):
        k = body.find("\n  | "); stmt, proof = body[:k], body[k+1:]
    else:
        stmt, proof = body[:k+2], body[k+3:]
    return stmt, proof

def size(proof):
    lines = sum(max(1, math.ceil(len(l) / 100)) for l in proof.split("\n"))
    chars = len(re.sub(r"\s", "", proof))
    return lines, chars
