/-- A code under `noneRead`: a level for the lane `true` and a level for the lane `false`. -/

/-- A configuration of a code's class goes by a move into the class of the moved code. For every

    decode (move c a) (Drain.next s a)

  | c, false, _, .inl rfl => Or.inl (congrArg (Prod.mk true) (Lv.val_lower c.1).symm)

/-- Each configuration of the moved code's class is reached by the move from the code's class.

  | c, false, _, .inr rfl => ⟨(false, c.2.val), Or.inr rfl, rfl⟩

/-- The level in lane `b` after the move `a`: kept where the move matches the lane, lowered
otherwise. -/
def after : Bool → Lv → Bool → Lv
  | true, k, true => k
  | false, k, false => k
  | true, k, false => k.lower
  | false, k, true => k.lower

/-- From lane `b` at level `k`, the move `a` leads to lane `b` at `after b k a`. No premise.
Calls `Lv.val_lower`. -/
theorem next_single : ∀ (b : Bool) (k : Lv) (a : Bool),
    Drain.next (b, k.val) a = (b, (after b k a).val)
  | true, _, true => rfl
  | false, _, false => rfl
  | true, k, false => congrArg (Prod.mk true) (Lv.val_lower k).symm
  | false, k, true => congrArg (Prod.mk false) (Lv.val_lower k).symm

/-- Whether a number is the number of a level, written out case by case. -/
def eqv : Lv → Nat → Bool
  | l0, 0 => true
  | l0, _ + 1 => false
  | l1, 0 => false
  | l1, 1 => true
  | l1, _ + 2 => false
  | l2, 0 => false
  | l2, 1 => false
  | l2, 2 => true
  | l2, _ + 3 => false

/-- Where `eqv k o` is `true`, `o` is the number of `k`. No premise. -/
theorem eqv_true : ∀ (k : Lv) (o : Nat), eqv k o = true → k.val = o
  | l0, 0, _ => rfl
  | l0, _ + 1, h => nomatch h
  | l1, 0, h => nomatch h
  | l1, 1, _ => rfl
  | l1, _ + 2, h => nomatch h
  | l2, 0, h => nomatch h
  | l2, 1, h => nomatch h
  | l2, 2, _ => rfl
  | l2, _ + 3, h => nomatch h

/-- Where `eqv k o` is `false`, `o` is not the number of `k`. No premise. Calls `succ_not_zero`
(written with `Nat.noConfusion`; `nomatch` on equations of numbers brought `propext`). -/
theorem eqv_false : ∀ (k : Lv) (o : Nat), eqv k o = false → k.val ≠ o
  | l0, 0, h => nomatch h
  | l0, n + 1, _ => fun e => succ_not_zero n e.symm
  | l1, 0, _ => fun e => succ_not_zero 0 e
  | l1, 1, h => nomatch h
  | l1, n + 2, _ => fun e => succ_not_zero n (congrArg Nat.pred e).symm
  | l2, 0, _ => fun e => succ_not_zero 1 e
  | l2, 1, _ => fun e => succ_not_zero 0 (congrArg Nat.pred e)
  | l2, 2, h => nomatch h
  | l2, n + 3, _ => fun e => succ_not_zero n (congrArg Nat.pred (congrArg Nat.pred e)).symm

/-- The code of one lane at one level, or the empty code, by a Boolean. -/
def pickB (b : Bool) (j : Lv) : Bool → LCode
  | true => .single b j
  | false => .empty

/-- For every Boolean `x` that is `true` only where `j.val = o` and `false` only where not, the
class of `pickB b j x` holds exactly lane `b` at `j` where `j.val = o`. No premise. -/
theorem decode_pickB (b : Bool) (j : Lv) (o : Nat) : ∀ x : Bool, (x = true → j.val = o) →
    (x = false → j.val ≠ o) → ∀ s, decodeL (pickB b j x) s ↔ (s = (b, j.val) ∧ j.val = o)
  | true, yes, _, _ => ⟨fun h => ⟨h, yes rfl⟩, fun h => h.1⟩
  | false, _, no, _ => ⟨fun h => False.elim h, fun h => absurd h.2 (no rfl)⟩

/-- What is seen after the move `a` from lane `b` at level `k`, the value kept being `o`, is
lane `b` at `after b k a`, where its number is `o`. No premise. Calls `next_single`. -/
theorem seen_single (b : Bool) (k : Lv) (a : Bool) (o : Nat) (t : Bool × Nat) :
    Seen Drain.process Drain.keepsLevel (decodeL (.single b k)) a o t ↔
      (t = (b, (after b k a).val) ∧ (after b k a).val = o) := by
  constructor
  · intro seen
    obtain ⟨s, held, step, lvl⟩ := seen
    have at_s : s = (b, k.val) := held
    subst at_s
    have reached : t = (b, (after b k a).val) := Eq.trans step (next_single b k a)
    exact ⟨reached, Eq.trans (congrArg Prod.snd reached).symm lvl⟩
  · intro h
    exact ⟨(b, k.val), rfl, Eq.trans h.1 (next_single b k a).symm,
      Eq.trans (congrArg Prod.snd h.1) h.2⟩

/-- What is seen after the move `a` from the two starts, the value kept being `o`: lane `a` at
level 2 where `o` is 2, and the other lane at level 1 where `o` is 1. No premise. -/
theorem seen_starts (a : Bool) (o : Nat) (t : Bool × Nat) :
    Seen Drain.process Drain.keepsLevel Drain.starts a o t ↔
      (t = (a, 2) ∧ 2 = o) ∨ (t = (!a, 1) ∧ 1 = o) := by
  cases a with
  | true =>
    constructor
    · intro seen
      obtain ⟨s, held, step, lvl⟩ := seen
      have reached : t = Drain.next s true := step
      have level : t.2 = o := lvl
      have held' : s = (true, 2) ∨ s = (false, 2) := held
      cases held' with
      | inl e =>
        subst e
        exact Or.inl ⟨reached, Eq.trans (congrArg Prod.snd reached).symm level⟩
      | inr e =>
        subst e
        exact Or.inr ⟨reached, Eq.trans (congrArg Prod.snd reached).symm level⟩
    · intro h
      cases h with
      | inl h => exact ⟨(true, 2), Or.inl rfl, h.1, Eq.trans (congrArg Prod.snd h.1) h.2⟩
      | inr h => exact ⟨(false, 2), Or.inr rfl, h.1, Eq.trans (congrArg Prod.snd h.1) h.2⟩
  | false =>
    constructor
    · intro seen
      obtain ⟨s, held, step, lvl⟩ := seen
      have reached : t = Drain.next s false := step
      have level : t.2 = o := lvl
      have held' : s = (true, 2) ∨ s = (false, 2) := held
      cases held' with
      | inl e =>
        subst e
        exact Or.inr ⟨reached, Eq.trans (congrArg Prod.snd reached).symm level⟩
      | inr e =>
        subst e
        exact Or.inl ⟨reached, Eq.trans (congrArg Prod.snd reached).symm level⟩
    · intro h
      cases h with
      | inl h => exact ⟨(false, 2), Or.inr rfl, h.1, Eq.trans (congrArg Prod.snd h.1) h.2⟩
      | inr h => exact ⟨(true, 2), Or.inl rfl, h.1, Eq.trans (congrArg Prod.snd h.1) h.2⟩

/-- The code after the move `a` from the two starts, by a Boolean saying whether `o` is 2. -/
def pickB2 (a : Bool) (o : Nat) : Bool → LCode
  | true => .single a l2
  | false => pickB (!a) l1 (eqv l1 o)

/-- For every Boolean `x` that is `true` only where `o` is 2 and `false` only where not, the
class of `pickB2 a o x` is what `seen_starts` says is seen. No premise. Calls `decode_pickB`,
`eqv_true`, `eqv_false`. -/
theorem decode_pickB2 (a : Bool) (o : Nat) : ∀ x : Bool, (x = true → 2 = o) →
    (x = false → 2 ≠ o) → ∀ t, decodeL (pickB2 a o x) t ↔
      ((t = (a, 2) ∧ 2 = o) ∨ (t = (!a, 1) ∧ 1 = o))
  | true, yes, _, _ => ⟨fun h => Or.inl ⟨h, yes rfl⟩, fun h => match h with
      | .inl h => h.1
      | .inr h => (succ_not_zero 0 (congrArg Nat.pred (Eq.trans (yes rfl) h.2.symm))).elim⟩
  | false, _, no, t =>
    (decode_pickB (!a) l1 o (eqv l1 o) (eqv_true l1 o) (eqv_false l1 o) t).trans
      ⟨fun h => Or.inr h, fun h => match h with
        | .inl h => absurd h.2 (no rfl)
        | .inr h => h⟩

/-- **D2, `updates`, in the form `ScratchViableEdge` stated it, for every code (stronger than the package's law asks).** For every code `c` and move `a`
usable from `decodeL c` with every move allowed, and every value `o : Nat` of
`Drain.keepsLevel`, the ones no configuration reaches included, some code decodes,

theorem updatesLevel : ∀ c a, Usable Drain.process everyMove (decodeL c) a → ∀ o,
    ∃ d, ∀ s, decodeL d s ↔ Seen Drain.process Drain.keepsLevel (decodeL c) a o s
  | .both, a, _, o => ⟨pickB2 a o (eqv l2 o), fun t =>
      (decode_pickB2 a o (eqv l2 o) (eqv_true l2 o) (eqv_false l2 o) t).trans
        (seen_starts a o t).symm⟩
  | .empty, _, _, _ => ⟨.empty, fun _ => ⟨fun h => False.elim h,
      fun seen => match seen with | ⟨_, h, _⟩ => h⟩⟩
  | .single b k, a, _, o => ⟨pickB b (after b k a) (eqv (after b k a) o), fun t =>
      (decode_pickB b (after b k a) o (eqv (after b k a) o) (eqv_true _ o) (eqv_false _ o)
        t).trans (seen_single b k a o t).symm⟩

/-- **D2 with the law.** Under `Drain.keepsLevel` with every move allowed and the guard

      InClosedFamily Drain.process Drain.keepsLevel Drain.above everyMove (decodeL c) :=
  coded_kept_iff_closed (decode := decodeL) (fun c _ => everyLevel c) (fun c _ => updatesLevel c)

/-- **D3, second half, through the law.** Under `Drain.keepsLevel` with every move allowed and
the guard `Drain.above`, the two starts together lie in a closed family. No premise. Calls
`level_kept_iff_closed` at the code `both`, whose class is `Drain.starts` by definition (its
direction from kept to closed), and the filed `Drain.together_kept_level` at eight steps. -/
theorem starts_in_closed_family_level :
    InClosedFamily Drain.process Drain.keepsLevel Drain.above everyMove Drain.starts :=
  (level_kept_iff_closed .both).mp (Drain.together_kept_level allLevel.length)

/-- **D3, second half, without the codes.** The same statement as
`starts_in_closed_family_level`, with the family written out: the classes within the two starts
and the classes in one lane above zero. Neither `every` nor `updates` is used. No premise. Calls
the filed `Drain.starts_within`, `Drain.level_lane`, `Drain.next_own`, and `Seen.mono`,
`usable_all`. -/
theorem starts_in_closed_family_level_direct :
    InClosedFamily Drain.process Drain.keepsLevel Drain.above everyMove Drain.starts :=
  ⟨fun C => (∀ s, C s → Drain.starts s) ∨ ∃ b, Drain.Lane b C,
   Or.inl (fun _ h => h),
   fun _ held s hs => match held with
     | .inl sub => Drain.starts_within s (sub s hs)
     | .inr ⟨_, lane⟩ => (lane s hs).2,
   fun _ held => match held with
     | .inl sub => ⟨true, usable_all _ true, fun o => Or.inr ⟨Drain.laneOf o,
         fun x seen => Drain.level_lane o x (Seen.mono sub seen)⟩⟩
     | .inr ⟨b, lane⟩ => ⟨b, usable_all _ b, fun _ => Or.inr ⟨b, fun t seen => match seen with
         | ⟨s, hs, step, _⟩ =>
           have stays : Drain.next s b = s :=
             Eq.subst (motive := fun z => Drain.next s z = s) (lane s hs).1 (Drain.next_own s)
           have same : t = s := Eq.trans step stays
           ⟨Eq.subst (motive := fun z => z.1 = b) (Eq.symm same) (lane s hs).1,
            Eq.subst (motive := fun z => z.2 ≠ 0) (Eq.symm same) (lane s hs).2⟩⟩⟩⟩

end ScratchDrainOnTheLaw
