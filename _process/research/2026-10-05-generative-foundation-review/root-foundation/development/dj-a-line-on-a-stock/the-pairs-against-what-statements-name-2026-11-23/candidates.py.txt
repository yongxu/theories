"""Step 2 of the frozen file: from the Lean print and the closed list of sides, every theorem whose statement names
notions of two different sides. AI-written (Claude Fable, board row DJ), 23 November 2026.
usage: python3 -B candidates.py SIDES.json OUT.json
SIDES.json: {"sides": {"P3.renewing": {"direct": [full names], "reading": [full names]}, ...}}
A statement names a side DIRECTLY where it names one of the side's notions, and THROUGH A DEFINITION where it names a
definition whose own type or value names one (one step, from the same print; a notion of any side is never counted
as a definition through which another side is named... it is counted only as itself)."""
import json, sys, collections, re
from pathlib import Path
ROOT = Path('/Users/ren/theories')
PKG = ROOT / 'experiments/integrated-correction/general-framework/generative-basis/lean/becoming'
def load():
    d = json.load(open(Path(__file__).parent / 'statements23.json'))
    receipt = json.load(open(PKG / 'becoming-verification.json'))
    return d, receipt
def source_of(receipt):
    """module -> the file it is in (the receipt's source paths are relative to the Lean tree's root)."""
    return {m: str(PKG.parent / q) for m, q in receipt['source_paths'].items()}
def main(sides_path, out_path):
    d, receipt = load()
    sides = json.load(open(sides_path))['sides']
    defs = {x['name']: x for x in d['notions']}
    side_of = collections.defaultdict(set)      # notion -> sides (direct)
    basis = {}
    for s, v in sides.items():
        for k in ('direct', 'reading'):
            for n in v.get(k, []):
                if n not in defs: raise SystemExit(f'not in the print: {n} ({s})')
                side_of[n].add(s); basis[(n, s)] = k
    listed = set(side_of)
    through = collections.defaultdict(set)      # definition -> sides it is built from (one step), if not itself listed
    for n, x in defs.items():
        if n in listed: continue
        if x['kind'] == 'constructor' or n.rsplit('.', 1)[-1].startswith('inst'): continue
        for u in x['names']:
            for s in side_of.get(u, ()): through[n].add((s, u))
    parts = json.load(open(sides_path)).get('parts_unfolded', {})
    part_of = collections.defaultdict(set)
    for s_, ns in parts.items():
        for n in ns: part_of[n].add(s_)
    controls = json.load(open(Path(__file__).parent / 'controls.json'))['controls'] if (Path(__file__).parent / 'controls.json').exists() else {}
    src = source_of(receipt); text = {}
    def stated(t):
        if t['line'] == 0: return False
        if t['module'] not in text: text[t['module']] = open(src[t['module']]).read().split('\n')
        seg = text[t['module']][t['line'] - 1:t['end']]
        return any(re.match(r'\s*(@\[[^\]]*\]\s*)?(private |protected )?(theorem|lemma)\b', l) for l in seg)
    rows = []; not_stated = 0; control_rows = []
    for t in d['theorems']:
        if not stated(t):
            not_stated += 1; continue
        def sides_in(names):
            direct = collections.defaultdict(list); thr = collections.defaultdict(list)
            for u in names:
                for s in side_of.get(u, ()): direct[s].append(u)
                for (s, base) in through.get(u, ()): thr[s].append(f'{u} (built from {base})')
            return direct, thr
        bd, bt = sides_in(t['in_binders']); cd, ct = sides_in(t['in_conclusion'])
        direct_sides = set(bd) | set(cd)
        all_sides = direct_sides | set(bt) | set(ct)
        part_sides = {s_ for u in t['in_binders'] + t['in_conclusion'] for s_ in part_of.get(u, ())}
        if t['name'] in controls:
            control_rows.append({'name': t['name'], 'named_in': controls[t['name']], 'direct_sides': sorted(direct_sides),
                                 'all_sides': sorted(all_sides), 'parts': sorted(part_sides - all_sides),
                                 'module': t['module'], 'line': t['line'], 'end': t['end']})
        if len(all_sides) < 2 and not (len(all_sides | part_sides) >= 2): continue
        if len(all_sides) < 2: all_sides = all_sides | part_sides
        rows.append({'control': t['name'] in controls,'name': t['name'], 'module': t['module'], 'line': t['line'], 'end': t['end'],
                     'premises_direct': {s: sorted(v) for s, v in bd.items()},
                     'conclusion_direct': {s: sorted(v) for s, v in cd.items()},
                     'premises_through': {s: sorted(v) for s, v in bt.items()},
                     'conclusion_through': {s: sorted(v) for s, v in ct.items()},
                     'direct_sides': sorted(direct_sides), 'all_sides': sorted(all_sides),
                     'level': 'direct' if len(direct_sides) >= 2 else ('through a definition' if len(set(bd) | set(cd) | set(bt) | set(ct)) >= 2 else 'through an unfolded part'),
                     'rests_on_a_reading': sorted({f'{u} as {s}' for s in direct_sides for u in (bd.get(s, []) + cd.get(s, []))
                                                   if basis.get((u, s)) == 'reading'})})
    pairs = collections.Counter(); pairs_direct = collections.Counter()
    for r in rows:
        ss = r['all_sides']
        for i in range(len(ss)):
            for j in range(i + 1, len(ss)):
                pairs[(ss[i], ss[j])] += 1
        ds = r['direct_sides']
        for i in range(len(ds)):
            for j in range(i + 1, len(ds)):
                pairs_direct[(ds[i], ds[j])] += 1
    names = sorted(sides)
    blank = [(a, b) for i, a in enumerate(names) for b in names[i + 1:] if pairs[(a, b)] == 0]
    naming = {s: sum(1 for t in d['theorems'] if any(s in side_of.get(u, ()) for u in t['in_binders'] + t['in_conclusion'])) for s in names}
    json.dump({'sides': sides, 'rows_not_stated_theorems': not_stated, 'controls': control_rows,
               'statements_naming_a_side_directly': naming,
               'candidates': rows,
               'pairs_direct': {f'{a} | {b}': c for (a, b), c in sorted(pairs_direct.items())},
               'pairs_direct_or_through': {f'{a} | {b}': c for (a, b), c in sorted(pairs.items())},
               'pairs_of_sides_no_statement_names_both': [f'{a} | {b}' for a, b in blank]},
              open(out_path, 'w'), indent=1, ensure_ascii=False)
    print(len(rows), 'candidates;', collections.Counter(r['level'] for r in rows), '; rows not stated theorems:', not_stated)
    def cell(r):
        ps = sorted({x.split('.')[0] for x in r['all_sides']})
        return ' x '.join(ps) if len(ps) > 1 else ps[0] + ' within'
    print('  by pair of pairs:', sorted(collections.Counter((cell(r), r['level']) for r in rows).items()))
    print('  controls:', len(control_rows), '; naming two sides:', sum(1 for c in control_rows if len(c['all_sides']) >= 2), '; one side:', sum(1 for c in control_rows if len(c['all_sides']) == 1), '; none:', sum(1 for c in control_rows if not c['all_sides']))
    print('  blank pairs:', len(blank), 'of', len(names) * (len(names) - 1) // 2)
if __name__ == '__main__':
    main(sys.argv[1], sys.argv[2])
