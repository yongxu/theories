#!/usr/bin/env python3
"""What links the atlas's dynamics would have if they were read off Lean's proof terms (a measurement, 22 November 2026).

AI-written (Claude Fable, board row DJ). Reads the atlas's store and the whole-package list of proof terms
(/Users/ren/tmp/dj/batch29/all-proof-heads.jsonl, printed by Lean on main 10742ca6f). A "proved-from" link X -> Y: a
theorem listed under dynamic X has in its proof term, directly or through theorems no dynamic lists, a theorem listed
under dynamic Y. It is a dependence, not a case and not a road; the list is one tool's output and a floor.
Run from the repository root. Prints counts and writes links-from-lean.json beside itself."""
import collections, json, os
HERE = os.path.dirname(os.path.abspath(__file__))
d = json.load(open('theories/becoming/results/atlas/entries/dynamics.json'))
rows = {}
for l in open('/Users/ren/tmp/dj/batch29/all-proof-heads.jsonl'):
    r = json.loads(l); rows[r['name']] = r
ids = [e['id'] for e in d]
home = collections.defaultdict(set)
for e in d:
    for n in e['declarations']:
        if isinstance(n, str):
            if n in rows: home[n].add(e['id'])
            elif 'Becoming.' + n in rows: home['Becoming.' + n].add(e['id'])
edges = collections.defaultdict(list)
for t, hs in home.items():
    seen, todo = set(), [(u, [t]) for u in rows[t]['uses']]
    while todo:
        u, path = todo.pop()
        if u in seen: continue
        seen.add(u)
        if u in home:
            for x in hs:
                for y in home[u]:
                    if x != y: edges[(x, y)].append(path + [u])
        elif u in rows:
            todo += [(v, path + [u]) for v in rows[u]['uses']]
out = [{'from': x, 'to': y, 'witnesses': sorted(w, key=len)[:3], 'count': len(w)} for (x, y), w in sorted(edges.items())]
json.dump({'main': '10742ca6f', 'list_sha256_16': '1f7f88bed4bb6307', 'links': out}, open(os.path.join(HERE, 'links-from-lean.json'), 'w'), ensure_ascii=False, indent=1)
print(len(out), 'directed proved-from links;', len({frozenset((o['from'], o['to'])) for o in out}), 'pairs')
