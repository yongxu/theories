```lean

/-- The position is ok and has a live move every outcome of which is in X. -/
def stayStep (ok : T → Prop) (live : T → A → Prop) (next : T → A → T → Prop)
    (X : T → Prop) (t : T) : Prop := ok t ∧ ∃ a, live t a ∧ ∀ u, next t a u → X u

def outStep (bad : T → Prop) (dead : T → A → Prop) (next : T → A → T → Prop)
```
