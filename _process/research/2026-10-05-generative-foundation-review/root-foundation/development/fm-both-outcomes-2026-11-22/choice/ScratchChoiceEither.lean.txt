import Becoming.ScratchChoice
import Becoming.Undecided

/-! AI-written scratch, board row FM. Finite searches over the replies and the moves. -/

namespace Becoming.Choice

universe u v

variable {T : Type u} {A : Type v} {ok bad : T → Prop} {live dead : T → A → Prop}
  {next : T → A → T → Prop} {moves : List A}
  (everyMove : ∀ a, a ∈ moves)
  (okOrBad : ∀ t, ok t ∨ bad t)
  (liveOrDead : ∀ t, ok t → ∀ a, live t a ∨ dead t a)

section Represented

variable {under : T → T → Prop}
  (outcomes : ∀ t, ok t → ∀ a, live t a → ∃ l : List T,
    (∀ u', u' ∈ l → next t a u' ∨ ¬ next t a u') ∧
    ∀ u, next t a u → ∃ u', u' ∈ l ∧ next t a u' ∧ under u u')

include everyMove okOrBad liveOrDead outcomes

/-- List every move, take ok or bad at each position (the horizon-zero alternative), and live or
dead for each move at ok positions (the horizon-one alternative for a sole move at an exclusively
ok position with exclusively staying outcomes). For each live move there, let a list decide
outcome membership and represent every outcome by a listed outcome above it under the relation.
If X or Y holds everywhere and X holds below each of its members, each position has a staying
step into X or an out step into Y. -/
theorem steps_either {X Y : T → Prop} (either : ∀ u, X u ∨ Y u)
    (down : ∀ u u', under u u' → X u' → X u) :
    ∀ t, stayStep ok live next X t ∨ outStep bad dead next Y t :=
  fun t => match okOrBad t with
    | .inr badHere => .inr (.inl badHere)
    | .inl safe =>
      let reply (a : A) := dead t a ∨ ∃ u, next t a u ∧ Y u
      let keeps (a : A) := live t a ∧ ∀ u, next t a u → X u
      have side (a : A) : reply a ∨ keeps a := match liveOrDead t safe a with
        | .inr no => .inl (.inl no)
        | .inl usable => match outcomes t safe a usable with
          | ⟨l, decided, cover⟩ =>
            have test (u : T) (mem : u ∈ l) : (next t a u → X u) ∨ (next t a u ∧ Y u) :=
              match decided u mem with
              | .inr no => .inl fun step => (no step).elim
              | .inl step => (either u).imp (fun x _ => x) (fun y => ⟨step, y⟩)
            match Undecided.each_or_some test with
            | .inl all => .inr ⟨usable, fun u step => match cover u step with
              | ⟨v, mem, reached, rel⟩ => down u v rel (all v mem reached)⟩
            | .inr ⟨u, _, reached, y⟩ => .inl (.inr ⟨u, reached, y⟩)
      match Undecided.each_or_some (l := moves) (fun a _ => side a) with
      | .inl replies => .inr (.inr fun a => replies a (everyMove a))
      | .inr ⟨a, _, keep⟩ => .inl ⟨safe, a, keep⟩

/-- List every move, cover positions by ok or bad and moves at ok positions by live or dead,
and represent outcomes of each live move there by listed outcomes above them with outcome
membership decided on the list. Suppose stage zero contains ok, each successor contains the
staying step of its predecessor, and every stage holds below its members. Each position is
in the stage or out at the same horizon. -/
theorem stage_either {stage : Nat → T → Prop} (zero : ∀ t, ok t → stage 0 t)
    (above : ∀ n t, stayStep ok live next (stage n) t → stage (n + 1) t)
    (down : ∀ n u u', under u u' → stage n u' → stage n u) :
    ∀ n t, stage n t ∨ Out bad dead next n t
  | 0, t => (okOrBad t).imp (zero t) id
  | n + 1, t => (steps_either everyMove okOrBad liveOrDead outcomes
      (stage_either zero above down n) (down n) t).imp (above n t) id

end Represented

variable (outcomes : ∀ t, ok t → ∀ a, live t a → ∃ l : List T,
  (∀ u', u' ∈ l → next t a u' ∨ ¬ next t a u') ∧
  ∀ u, next t a u → ∃ u', u' ∈ l ∧ next t a u' ∧ u = u')

include everyMove okOrBad liveOrDead outcomes

/-- List every move, cover positions by ok or bad and moves at ok positions by live or dead,
and list all outcomes of each live move there with outcome membership decided on its list.
Each position stays or is out at every horizon. -/
theorem stays_or_out : ∀ n t, Stays ok live next n t ∨ Out bad dead next n t :=
  stage_either everyMove okOrBad liveOrDead outcomes (fun _ => id) (fun _ _ => id)
    (fun _ _ _ same h => same.symm ▸ h)

variable (apart : ∀ t, ok t → bad t → False) (exclusive : ∀ t a, live t a → dead t a → False)

include apart exclusive

/-- List every move, cover positions by ok or bad and moves at ok positions by live or dead,
and list all outcomes of each live move there with membership as an outcome decided on its list.
Suppose ok excludes bad, live excludes dead and every ok position is listed. At t membership
in a stay-closed set and being out at the position-list length are exhaustive and mutually exclusive. -/
theorem closed_or_out {all : List T} (within : ∀ t, ok t → t ∈ all) {t : T} :
    ((∃ K, StayClosed ok live next K ∧ K t) ∨ Out bad dead next all.length t) ∧
    ¬ ((∃ K, StayClosed ok live next K ∧ K t) ∧ Out bad dead next all.length t) :=
  ⟨(stays_or_out everyMove okOrBad liveOrDead outcomes all.length t).imp
      (stays_iff_closed within).mp id,
   fun ⟨⟨_, closed, held⟩, out⟩ => closed_never_out apart exclusive closed held all.length out⟩

/-- List every move, cover positions by ok or bad and moves at ok positions by live or dead,
and list all outcomes of each live move there with outcome membership decided on its list.
If ok excludes bad, live excludes dead and every ok position is listed, being out at any horizon
implies being out at the position-list length. -/
theorem out_at_ok_length {all : List T} (within : ∀ t, ok t → t ∈ all) {t : T} :
    ∀ n, Out bad dead next n t → Out bad dead next all.length t :=
  fun n out => (closed_or_out everyMove okOrBad liveOrDead outcomes apart exclusive within).1.elim
    (fun ⟨_, closed, held⟩ => (closed_never_out apart exclusive closed held n out).elim) id

/-- List every move, cover positions by ok or bad and moves at ok positions by live or dead,
and list all outcomes of each live move there with outcome membership decided on its list.
If ok excludes bad and live excludes dead, being out at n is equivalent to being unable to
stay for n steps. -/
theorem out_iff_not_stays (n : Nat) (t : T) :
    Out bad dead next n t ↔ ¬ Stays ok live next n t :=
  ⟨fun out h => stays_disjoint apart exclusive n t h out,
   fun no => (stays_or_out everyMove okOrBad liveOrDead outcomes n t).elim (fun h => (no h).elim) id⟩

end Becoming.Choice
