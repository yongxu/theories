"""AI-written (Claude Fable, reader 'program'), 22 November 2026.

Question 6: the Drain case, recomputed with the literal model (naive.py).

ScratchKeptForViable.lean l. 265-281:
    next (true, k) true = (true, k); next (false, k) false = (false, k)
    next (true, k) false = (true, k - 1); next (false, k) true = (false, k - 1)   (Nat: 0 - 1 = 0)
    process := <fun s a t => t = next s a>;  above s := s.2 != 0
    starts s := s = (true, 2) \\/ s = (false, 2);  keepsLevel _ _ t := t.2
    noneRead _ _ _ := ();  everyMove _ _ := True
ScratchDrainOnTheLaw.lean l. 58-73, 213-227:
    Code := Lv x Lv; decode c s := s = (true, c.1.val) \\/ s = (false, c.2.val); nine codes
    LCode := both | empty | single b k; decodeL both = starts, empty = nothing,
    single b k = {(b, k.val)}; eight codes

The Lean configurations are Bool x Nat. No move raises a level and every coded class has levels
at most 2, so the classes seen from coded classes stay at levels at most 2. The program takes
levels 0..LMAX (default 3, one more than is reached) and values 0..LMAX+2 for the level reading;
it is run for LMAX = 2 and LMAX = 3 to show the extra level changes nothing.
"""
import sys
from naive import Model

LMAX = int(sys.argv[1]) if len(sys.argv) > 1 else 3
S = [(b, k) for b in (True, False) for k in range(LMAX + 1)]
A = [False, True]


def nxt(s, a):
    b, k = s
    return (b, k) if a == b else (b, max(k - 1, 0))


step = {(s, a, nxt(s, a)) for s in S for a in A}
guard = {s for s in S if s[1] != 0}
allowed = {(s, a) for s in S for a in A}
print("levels 0..%d, %d configurations" % (LMAX, len(S)))

# --- the reading that keeps nothing
codes = {(i, j): {(True, i), (False, j)} for i in range(3) for j in range(3)}
M = Model(S, A, ["()"], step, {e: "()" for e in step}, guard, allowed, codes)
print("\n-- noneRead: nine codes; updates (guarded, usable):", M.updates(True),
      "; updates (every code, move):", M.updates(False))
print("   every code guarded or with a configuration outside:",
      all(M.codeGuard(c) or M.Outside(c) for c in M.C),
      "; guarded codes:", sorted(c for c in M.C if M.codeGuard(c)))
print("   every move usable from every coded class:",
      all(M.Usable(M.decode[c], a) for c in M.C for a in A))
start = (2, 2)
print("   Lost n (l2,l2), n = 0..10:", [int(M.Lost(n, start)) for n in range(11)])
print("   KeptFor n starts, n = 0..10:", [int(M.KeptFor(n, M.decode[start])) for n in range(11)])
first = {c: next((n for n in range(12) if M.Lost(n, c)), None) for c in M.C}
print("   least losing level of each code:", first)
print("   Lost 9 (l2,l2):", M.Lost(9, start), "; starts in a closed family:",
      M.InClosedFamily(M.decode[start]))
print("   no code is both lost and kept at a level 0..10:",
      not any(M.Lost(n, c) and M.KeptFor(n, M.decode[c]) for c in M.C for n in range(11)),
      "; every code is one or the other at each level 0..10:",
      all(M.Lost(n, c) or M.KeptFor(n, M.decode[c]) for c in M.C for n in range(11)))
# the three-move argument of Drain.losing, recomputed
worst = []
for a1 in A:
    for a2 in A:
        for a3 in A:
            ends = [nxt(nxt(nxt((b, 2), a1), a2), a3) for b in (True, False)]
            worst.append(min(e[1] for e in ends))
print("   over the 8 lists of three moves, the lower of the two levels reached:", worst)
two = []
for a1 in A:
    for a2 in A:
        ends = [nxt(nxt((b, 2), a1), a2) for b in (True, False)]
        two.append(((a1, a2), min(e[1] for e in ends)))
print("   over the 4 lists of two moves:", two)

# --- the reading that keeps the level reached
O = list(range(LMAX + 3))
lcodes = {"both": {(True, 2), (False, 2)}, "empty": set()}
for b in (True, False):
    for k in range(3):
        lcodes[("single", b, k)] = {(b, k)}
ML = Model(S, A, O, step, {e: e[2][1] for e in step}, guard, allowed, lcodes)
print("\n-- keepsLevel: eight codes, values 0..%d; updates (guarded, usable): %s; "
      "updates (every code, move): %s" % (O[-1], ML.updates(True), ML.updates(False)))
print("   Lost n both, n = 0..12:", [int(ML.Lost(n, "both")) for n in range(13)])
print("   KeptFor n starts, n = 0..12:", [int(ML.KeptFor(n, lcodes["both"])) for n in range(13)])
print("   starts in a closed family:", ML.InClosedFamily(lcodes["both"]))
firstL = {str(c): next((n for n in range(13) if ML.Lost(n, c)), None) for c in ML.C}
print("   least losing level of each code (None = not lost at 0..12):", firstL)
print("   every code one or the other at each level 0..12:",
      all(ML.Lost(n, c) or ML.KeptFor(n, ML.decode[c]) for c in ML.C for n in range(13)),
      "; never both:",
      not any(ML.Lost(n, c) and ML.KeptFor(n, ML.decode[c]) for c in ML.C for n in range(13)))
print("   distinct classes seen per (code, move), over the values 0..49:")
BIG = range(50)
worst = 0
for c in ML.C:
    for a in A:
        by = {}
        for o in BIG:
            seen = frozenset(t for t in S if any(s in ML.decode[c] and (s, a, t) in step
                                                 and t[1] == o for s in S))
            by.setdefault(seen, []).append(o)
        worst = max(worst, len(by))
        desc = "; ".join("%s at %s" % (sorted(k), v if len(v) <= 3 else "the other %d values" % len(v))
                         for k, v in by.items())
        print("     %-22s move %-5s: %d classes: %s" % (c, a, len(by), desc))
print("   most distinct classes seen at one (code, move):", worst)
# would a fixed short list of values stand for every value?
for values in ([0, 1, 2], [1, 2, 3], [0, 1, 2, 3]):
    ok = True
    for c in ML.C:
        for a in A:
            listed = {frozenset(t for t in S if any(s in ML.decode[c] and (s, a, t) in step
                                                    and t[1] == o for s in S)) for o in values}
            for o in BIG:
                seen = frozenset(t for t in S if any(s in ML.decode[c] and (s, a, t) in step
                                                     and t[1] == o for s in S))
                if seen not in listed:
                    ok = False
    print("   the list %s gives, at every (code, move), every class seen at a value 0..49: %s"
          % (values, ok))

# a short list of values in place of every value: the same kept and lost levels?
ref = {(n, str(c)): (ML.KeptFor(n, ML.decode[c]), ML.Lost(n, c)) for c in ML.C for n in range(13)}
for values in ([0, 1, 2], [1, 2, 3], [1, 2]):
    MQ = Model(S, A, values, step, {e: e[2][1] for e in step}, guard, allowed, lcodes)
    same = all(ref[(n, str(c))] == (MQ.KeptFor(n, MQ.decode[c]), MQ.Lost(n, c))
               for c in MQ.C for n in range(13))
    diff = [(str(c), n) for c in MQ.C for n in range(13)
            if ref[(n, str(c))] != (MQ.KeptFor(n, MQ.decode[c]), MQ.Lost(n, c))][:4]
    print("   with only the values %s: kept and lost levels of the eight codes at n = 0..12 "
          "the same as with all values: %s %s" % (values, same, diff))
