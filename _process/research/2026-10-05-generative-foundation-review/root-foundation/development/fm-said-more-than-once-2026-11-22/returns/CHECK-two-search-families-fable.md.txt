```lean
theorem first_below {P Q : Nat → Prop} :
    ∀ n, (∀ t, t < n → P t ∨ Q t) →
      (∀ t, t < n → P t) ∨ ∃ m, m < n ∧ Q m ∧ ∀ t, t < m → P t
  | 0, _ => .inl fun _ lt => absurd lt (Nat.not_lt_zero _)
  | n + 1, decided =>
    match first_below n fun t lt => decided t (Nat.lt_succ_of_lt lt) with
    | .inr ⟨m, lt, fails, before⟩ => .inr ⟨m, Nat.lt_succ_of_lt lt, fails, before⟩
    | .inl all => match decided n (Nat.lt_succ_self n) with
      | .inr fails => .inr ⟨n, Nat.lt_succ_self n, fails, all⟩
      | .inl holds => .inl fun t lt => match Nat.lt_or_ge t n with
        | .inl before => all t before
        | .inr after =>
          Eq.subst (motive := P) (Nat.le_antisymm (Nat.le_of_lt_succ lt) after).symm holds
```
