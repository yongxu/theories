def row_index(mod):
    hits = [k for k, l in enumerate(lines) if l.startswith('| L1 | [`Becoming.%s`]' % mod) or l.startswith('| L0 | [`Becoming.%s`]' % mod)]
    assert len(hits) == 1, (mod, hits); return hits[0]
R1 = ("| L1 | [`Becoming.PredicateIteration`](Becoming/PredicateIteration.lean) | a step on predicates applied as many times as a list is long (imports only `Becoming.Prelude`; no premise beyond what each statement names): for a step that preserves inclusion and whose value at the start lies inside the start, where a list holds every state of the start, the iterate at the list's length lies inside the next, is the iterate at every length, and is the largest class inside the start that lies inside its own step (`Iteration.iter_persists`, `Iteration.iter_iff_all`, `Iteration.iter_largest`); for a step whose start lies inside its own first step, with G the proposition that a state y is in the iterate at a list's length: if the step commutes with implication to G (`Iteration.CommutesWithImplication`) and the list holds every state at which membership in the start implies G, then y in any iterate is in the iterate at the list's length (`Iteration.iter_within_length`); two lemmas on lists (`remove_mem`, `gather_listed`) |")
R2 = ("| L1 | [`Becoming.RisingIteration`](Becoming/RisingIteration.lean) | rising iteration where every state is listed (through `Becoming.PredicateIteration` and `Becoming.Decidedness`): every monotone step then commutes with implication to falsity (`Iteration.commutes_false`), so for a step whose start lies inside its own first step a state in the iterate one past the list's length cannot be refuted to be in the iterate at its length, and is in it where that follows from its double negation (`Iteration.rise_not_not`, `Iteration.rise_of_stable`); if, for every monotone step on a one-state type, being in iterate 2 from the empty predicate implied being in iterate 1, every proposition would be decided (`Iteration.singleton_rise_needs_em`) |")

import json
reg = wt/'theories/becoming/results/data/regions.json'
raw = reg.read_text()
d = json.loads(raw)
def dump(x): return json.dumps(x, ensure_ascii=False, indent=1) + ('\n' if raw.endswith('\n') else '')
assert dump(d) == raw, 'regions.json does not round-trip; edit it by text'
mods = d['modules']
was_sorted = list(mods) == sorted(mods)
mods['Becoming.PredicateIteration'] = {"region": "method", "base": "general",
    "note": "a monotone step on predicates applied as many times as a list is long: the iterate at the list's length is every iterate; rising steps that commute with implication"}
mods['Becoming.RisingIteration'] = {"region": "method", "base": "general",
    "note": "rising iteration where every state is listed: up to a double negation; what a bound for every step would decide"}
if was_sorted: d['modules'] = {k: mods[k] for k in sorted(mods)}
reg.write_text(dump(d))
