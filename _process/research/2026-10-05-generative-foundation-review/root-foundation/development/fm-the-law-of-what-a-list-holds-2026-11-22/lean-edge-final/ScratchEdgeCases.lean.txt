section BoundedReach

variable {S : Type u} {A : Type v} {P : Process S A} {B : S → Prop}
local notation "W" => iter (reachStep (Steps P) B) B

/-- Reaching a target by level k is exactly having a trace of at most k moves into it. -/
theorem rise_iff_trace : ∀ k y, W k y ↔ ∃ t as, as.length ≤ k ∧ Trace P y as t ∧ B t :=
  iter_iff_stages (stage := fun k y => ∃ t as, as.length ≤ k ∧ Trace P y as t ∧ B t) reach_mono
    (fun y => ⟨fun hit => ⟨y, [], Nat.le_refl _, .nil y, hit⟩, fun
      | ⟨_, [], _, .nil _, hit⟩ => hit
      | ⟨_, _ :: _, le, _, _⟩ => absurd le (Nat.not_succ_le_zero _)⟩)
    (fun _ y => ⟨fun
      | .inl hit => ⟨y, [], Nat.zero_le _, .nil y, hit⟩
      | .inr ⟨_, ⟨a, st⟩, t, as, le, path, hit⟩ => ⟨t, a :: as, Nat.succ_le_succ le, .cons st path, hit⟩,
      fun
      | ⟨_, [], _, .nil _, hit⟩ => .inl hit
      | ⟨t, a :: as, le, .cons st path, hit⟩ => .inr ⟨_, ⟨a, st⟩, t, as, Nat.le_of_succ_le_succ le, path, hit⟩⟩)

variable {all : List S} (every : ∀ x, x ∈ all)

include every in
/-- If a list of length n + 1 holds every state, each reachable target has a trace to it of at most n moves. -/
theorem reach_within_count {n : Nat} (len : all.length = n + 1) {s y : S} (reach : Reach P y s) :
    ∃ as, as.length ≤ n ∧ Trace P y as s :=
  match (rise_iff_trace (B := fun t => t = s) n y).mp
      ((closure_iff_reach_within every len).mp ⟨s, reach_iff_withinClosure.mp reach, rfl⟩) with
  | ⟨_, as, le, path, rfl⟩ => ⟨as, le, path⟩

include every in
/-- If every state is listed and k is at least the predecessor of the list length,
each reachable target has a trace to it of at most k moves. -/
theorem reach_within_of_pred_le {k : Nat} (enough : all.length.pred ≤ k) {s y : S} (reach : Reach P y s) :
    ∃ as, as.length ≤ k ∧ Trace P y as s :=
  (reach_within_count every (Nat.succ_pred_eq_of_pos (length_pos_of_mem (every y))).symm reach).imp
    fun _ h => ⟨Nat.le_trans h.1 enough, h.2⟩

/-- Suppose every state is listed, a Boolean test recognizes exactly the single steps, and the target
returns. Every state reaching the target has a trace to it of at most the list length. -/
theorem reach_within_listed_case {s : S} {all : List S} {rel : S → S → Bool} (every : ∀ x, x ∈ all)
    (decides : ∀ x y, rel x y = true ↔ ∃ a, P.step x a y) (returns : Branching.Returns P s) {y : S}
    (reach : Reach P y s) : ∃ as, as.length ≤ all.length ∧ Trace P y as s :=
  let _ := And.intro decides returns; reach_within_of_pred_le every (Nat.pred_le _) reach

/-- The listed trace bound has the same statement as the one in Junctures at every pair of universes. -/
theorem reach_within_same : @reach_within_listed_case.{u,v} = @Branching.reach_within_listed.{u,v} := rfl

-- home: ReachableRanges
include every in
/-- If every state is listed and k is at least the predecessor of the list length,
an item is possible exactly when it is possible within k moves. -/
theorem possible_iff_within_count {X : Type} {adm : S → X → Prop} {y : S} {x : X} {k : Nat}
    (enough : all.length.pred ≤ k) : Reaching.Possible P adm y x ↔ Reaching.PossibleWithin P adm k y x :=
  ⟨fun ⟨t, reach, hit⟩ => match reach_within_of_pred_le every enough reach with
    | ⟨as, le, path⟩ => ⟨t, as, le, path, hit⟩, Reaching.PossibleWithin.possible⟩

end BoundedReach

section Matching

variable {S : Type u} {S' : Type i} {A : Type v} {C : Type w} {F : Frame S A C} {G : Frame S' A C}

/-- The pair agrees on claims and each step on either side has a same-move match
whose pair of outcomes lies in the given class. -/
def alikeStep (F : Frame S A C) (G : Frame S' A C) (X : S × S' → Prop) (p : S × S') : Prop :=
  AlikeTo F G 0 p.1 p.2 ∧
    (∀ a t, F.P.step p.1 a t → ∃ t', G.P.step p.2 a t' ∧ X (t, t')) ∧
    (∀ a t', G.P.step p.2 a t' → ∃ t, F.P.step p.1 a t ∧ X (t, t'))

/-- Enlarging the class of outcome pairs preserves each match in both directions. -/
theorem alike_mono : MonotoneStep (alikeStep F G) :=
  fun sub _ ⟨agree, forth, back⟩ => ⟨agree,
    fun a t st => match forth a t st with | ⟨t', st', h⟩ => ⟨t', st', sub _ h⟩,
    fun a t' st' => match back a t' st' with | ⟨t, st, h⟩ => ⟨t, st, sub _ h⟩⟩

local notation "agree" => (fun p : S × S' => AlikeTo F G 0 (Prod.fst p) (Prod.snd p))

/-- Iteration from agreement on claims is likeness to the same depth. -/
theorem iter_iff_alikeTo : ∀ n (p : S × S'), iter (alikeStep F G) agree n p ↔ AlikeTo F G n p.1 p.2 :=
  iter_iff_stages (stage := fun n p => AlikeTo F G n p.1 p.2) alike_mono
    (fun _ => Iff.rfl) (fun _ _ => Iff.rfl)

variable {pairs : List (S × S')}

/-- If every pair agreeing on claims is listed, likeness at the pair count persists one step. -/
theorem alike_at_cover (within : ∀ p, agree p → p ∈ pairs) (s : S) (s' : S') :
    AlikeTo F G pairs.length s s' → AlikeTo F G (pairs.length + 1) s s' :=
  fun h => (iter_iff_alikeTo _ (s, s')).mp (iter_persists alike_mono (fun _ m => m.1) within
    ((iter_iff_alikeTo _ (s, s')).mpr h))

/-- If every pair agreeing on claims is listed, likeness is equivalent to likeness at its count. -/
theorem alike_iff_at_cover (within : ∀ p, agree p → p ∈ pairs) {s : S} {s' : S'} :
    Alike F G s s' ↔ AlikeTo F G pairs.length s s' :=
  ⟨fun h => h.alikeTo _, fun h => ⟨_, AlikeTo.bisimulation (alike_at_cover within), h⟩⟩

variable {l : List S} {l' : List S'} (all : ∀ s, s ∈ l) (all' : ∀ s', s' ∈ l')
local notation "N" => l'.length * l.length

include all all' in
/-- With both configuration spaces listed, likeness persists at the product of the lengths. -/
theorem alike_at_pairs (s : S) (s' : S') : AlikeTo F G N s s' → AlikeTo F G (N + 1) s s' :=
  length_pairsOf l' l ▸ alike_at_cover (F := F) (G := G)
    (fun p _ => mem_pairsOf (all p.1) (all' p.2)) s s'

/-- With all moves, claims and configurations listed and both frames' steps and claims decided,
likeness at some depth at most the product of the configuration counts implies likeness one step later. -/
theorem steady_within_case (moves : ∃ l : List A, ∀ a, a ∈ l) (claims : ∃ l : List C, ∀ c, c ∈ l)
    (finite : FiniteDecided F) (finite' : FiniteDecided G) {l : List S} {l' : List S'}
    (all : ∀ s, s ∈ l) (all' : ∀ s', s' ∈ l') :
    ∃ k, k ≤ l'.length * l.length ∧ ∀ s s', AlikeTo F G k s s' → AlikeTo F G (k + 1) s s' :=
  let _ := And.intro moves (And.intro claims (And.intro finite finite'))
  ⟨_, Nat.le_refl _, alike_at_pairs all all'⟩

/-- The replacement for the existential depth bound has the package statement at all universes. -/
theorem steady_within_same : @steady_within_case.{u,v,w,i} = @AlikeTo.steady_within.{u,v,w,i} := rfl

include all all' in
/-- With both configuration spaces listed, likeness is equivalent to likeness at the product bound. -/
theorem alike_iff_alikeTo_pairs {s : S} {s' : S'} : Alike F G s s' ↔ AlikeTo F G N s s' :=
  length_pairsOf l' l ▸ alike_iff_at_cover (F := F) (G := G) (s := s) (s' := s')
    (fun p _ => mem_pairsOf (all p.1) (all' p.2))

end Matching

section Predecessors

variable {S : Type u} {A : Type v} {P : Process S A}

/-- A state has a predecessor in the given class by some move of the process. -/
def predStep (P : Process S A) (X : S → Prop) (s : S) : Prop := ∃ t, Steps P t s ∧ X t

/-- Enlarging the predecessor class preserves a witnessed step from it. -/
theorem pred_mono : MonotoneStep (predStep P) :=
  fun sub _ ⟨t, st, h⟩ => ⟨t, st, sub t h⟩

/-- Iterating the predecessor step from all states gives exactly the endpoints of
traces of that length, with freely chosen starts. -/
theorem iter_iff_reachable : ∀ n s, iter (predStep P) (fun _ => True) n s ↔ Branching.Reachable P n s :=
  iter_iff_stages pred_mono (fun s => ⟨fun _ => ⟨s, [], rfl, .nil s⟩, fun _ => trivial⟩)
    (fun _ s => ⟨fun ⟨t, ⟨a, st⟩, x, as, len, path⟩ => ⟨x, as ++ [a],
       (ListTests.length_append as [a]).trans (congrArg Nat.succ len), trace_append_iff.mpr ⟨t, path, .cons st (.nil s)⟩⟩,
     fun
       | ⟨_, [], len, _⟩ => Nat.noConfusion len
       | ⟨x, a :: as, len, path⟩ =>
         have split := List.dropLast_concat_getLast (List.cons_ne_nil a as)
         match trace_append_iff.mp (split.symm ▸ path : Trace P x (_ ++ [_]) s) with
         | ⟨t, earlier, final⟩ => ⟨t, ⟨_, trace_singleton_iff.mp final⟩, x, _, Nat.succ.inj
             ((ListTests.length_append _ [_]).symm.trans ((congrArg List.length split).trans len)), earlier⟩⟩)

variable {all : List S} (every : ∀ x, x ∈ all) {s : S}

include every in
/-- With all states listed, an endpoint is reachable at the list length exactly when
it is reachable at every exact length. -/
theorem reachable_iff_all : Branching.Reachable P all.length s ↔ ∀ n, Branching.Reachable P n s :=
  ⟨fun h n => (iter_iff_reachable n s).mp ((iter_iff_all pred_mono (fun _ _ => trivial)
      (fun x _ => every x)).mp ((iter_iff_reachable _ s).mpr h) n), fun h => h all.length⟩

variable {next : S → S}

include every in
/-- With every state of a determined process listed, a state returns exactly when it is reachable at the count. -/
theorem returns_iff_reachable_count : Law.Returns next s ↔ Law.Reachable next all.length s :=
  (Law.returns_iff_always_reachable every).trans (reachable_iff_all every).symm

/-- With every state of a determined process listed, a state is drawn exactly when it fails reachability at the count. -/

section Inquiry

variable {S : Type u} {R E : «际» S}

/-- The pair lies in E and each step on either side has a match whose pair of outcomes
lies in the given class. -/
def inquiryStep (R E : «际» S) (X : S × S → Prop) (p : S × S) : Prop :=
  E p.1 p.2 ∧ (∀ v, R p.1 v → ∃ v', R p.2 v' ∧ X (v, v')) ∧
    (∀ v', R p.2 v' → ∃ v, R p.1 v ∧ X (v, v'))

/-- Enlarging the class of outcome pairs preserves both matches of the inquiry step. -/
theorem inquiry_mono : MonotoneStep (inquiryStep R E) :=
  fun sub _ ⟨e, forth, back⟩ => ⟨e,
    fun v st => match forth v st with | ⟨v', st', h⟩ => ⟨v', st', sub _ h⟩,
    fun v' st' => match back v' st' with | ⟨v, st, h⟩ => ⟨v, st, sub _ h⟩⟩

/-- Iterating the inquiry step from E agrees with every grammar stage of inquiry. -/
theorem iter_iff_inquiry : ∀ n (p : S × S),
    iter (inquiryStep R E) (fun z => E z.1 z.2) n p ↔ «诘» R E n p.1 p.2 :=
  iter_iff_stages (stage := fun n p => «诘» R E n p.1 p.2) inquiry_mono
    (fun _ => Iff.rfl) (fun _ _ => Iff.rfl)

variable {all : List (S × S)} (within : ∀ p : S × S, E p.1 p.2 → p ∈ all) {x y : S}

include within in
/-- If every E-related pair is listed, a pair passes inquiry at the list length exactly
when it passes at every stage. -/
theorem inquiry_iff_all : «诘» R E all.length x y ↔ ∀ k, «诘» R E k x y :=
  ⟨fun h k => (iter_iff_inquiry k (x, y)).mp ((iter_iff_all inquiry_mono (fun _ m => m.1) within).mp
      ((iter_iff_inquiry _ (x, y)).mpr h) k), fun h => h all.length⟩

include within in
/-- If E is symmetric and every E-related pair is listed, the core is inquiry at the list length. -/
theorem core_iff_inquiry (sym : «则» E («逆» E)) : «核» R E x y ↔ «诘» R E all.length x y :=
  let D := «诘» R E all.length
  have levels (a b : S) (h : D a b) := (inquiry_iff_all within).mp h
  ⟨fun h => «核则不可致诘» h _, fun h =>
    ⟨D, fun a b ab => levels a b ab 0, «诘之对称» sym all.length,
     («有常之逐点» R D).mpr (fun a b ab => (levels a b ab (all.length + 1)).2.1), h⟩⟩

end Inquiry

section Keepable

open Viability
variable {S : Type u} {A : Type v} {P : Process S A} {g : S → Prop}

/-- The guard holds and some move has an outcome in the given class. -/
def keepStep (P : Process S A) (g X : S → Prop) (s : S) : Prop := g s ∧ courseStep P X s

/-- Enlarging the class of possible outcomes preserves the guarded course step. -/
theorem keep_mono : MonotoneStep (keepStep P g) :=
  fun sub s h => ⟨h.1, course_mono P sub s h.2⟩

/-- With every state listed, a run can stay in the guard from s exactly when s survives
the guarded course step for the list length. -/
theorem keepable_iff_count {all : List S} (every : ∀ x, x ∈ all) {s : S} :
    Keepable P g s ↔ iter (keepStep P g) g all.length s :=
  match iter_largest (keep_mono (P := P) (g := g)) (fun _ h => h.1) (fun x _ => every x) with
  | ⟨inside, post, largest⟩ =>
    ⟨largest (Keepable P g) (fun _ => Keepable.start) (fun _ => keepable_step_iff.mp) s,
     fun h => match Onward.KeepsMoving.run (P := P) every (fun t ht => (post t ht).2) h with
       | ⟨r, start, held⟩ => ⟨r, start, fun n => inside _ (held n)⟩⟩

end Keepable

section Tolerating
open Tolerance
variable {S : Type u}

/-- A step is monotone exactly when it keeps inclusion of predicates. -/
theorem monotone_iff_keepsInclusion (Φ : (S → Prop) → S → Prop) : MonotoneStep Φ ↔ KeepsInclusion Φ :=
  ⟨fun h X Y => @h X Y, fun h {X Y} => h X Y⟩

/-- For any step on predicates and any start, iterate n equals the predicate at time n of its orbit. -/
theorem iter_eq_orbit (Φ : (S → Prop) → S → Prop) (V₀ : S → Prop) : ∀ n, iter Φ V₀ n = Law.orbit Φ V₀ n
  | 0 => rfl
  | n + 1 => congrArg Φ (iter_eq_orbit Φ V₀ n)

/-- Suppose G is contained in the start and, for every member a, all but at most tol a entries in
its list belong to G. Every member of G then belongs to every stage of the tolerating process. -/
theorem kept_set_case {tol : S → Nat} {ups : S → List S} {G s : S → Prop} (start : ∀ a, G a → s a)
    (inside : ∀ a, G a → AllBut G (tol a) (ups a)) :
    ∀ n a, G a → (Law.run (tolerating tol ups) s).state n a :=
  fun n => Eq.subst (motive := fun X => ∀ a, G a → X a) (iter_eq_orbit _ s n)
    (postfixed_in_iter ((monotone_iff_keepsInclusion _).mpr (tolerating_keeps tol ups)) start inside n)

/-- The statement about a preserved set agrees with the one in Tolerance at every universe. -/
theorem kept_set_same : @kept_set_case.{u} = @Tolerance.kept_set.{u} := rfl
end Tolerating
