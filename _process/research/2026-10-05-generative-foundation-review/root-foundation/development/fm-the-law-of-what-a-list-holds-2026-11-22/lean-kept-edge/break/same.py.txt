# the Heyting check of the FALLING law on one point (it is proved in Lean; a check of the model)
from weaker import opens
for name, points, below in (("line", ["now", "later"], {"now": {"now", "later"}, "later": {"later"}}),
                            ("fork", ["now", "l", "r"], {"now": {"now", "l", "r"}, "l": {"l"}, "r": {"r"}})):
    H = opens(points, below)
    imp = lambda a, b: frozenset(x for x in points if all((y not in a) or (y in b) for y in below[x]))
    starts = fails = 0
    for vals in itertools.product(H, repeat=len(H)):
        F = dict(zip(H, vals))
        if not all(imp(X, Y) <= imp(F[X], F[Y]) for X in H for Y in H): continue
        for V0 in H:
            if F[V0] <= V0:
                starts += 1
                if not F[V0] <= F[F[V0]]: fails += 1
    print(f"falling law on one point in the undecided model '{name}': falling starts {starts}, failures {fails}")
