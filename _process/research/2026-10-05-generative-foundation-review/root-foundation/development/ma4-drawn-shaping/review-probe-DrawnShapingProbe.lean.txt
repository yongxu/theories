theorem one_pos : 0 < 1 := Nat.zero_lt_one

def even : CarrierCounts := ⟨1, 1, Nat.zero_lt_one⟩

/-! ## 1. `lo ≤ hi` is not idle where the module docstring drops it -/

/-- `c ≤ lo` without `lo ≤ hi` does not give favouring at every share (a = 1, p = 0, c = 1,
hi = 0, lo = 1: at one to one each shaper leaves three, each other four). -/
theorem everywhere_needs_step :
    (drawn 1 0 1 0 1 one_pos even).carrier < (drawn 1 0 1 0 1 one_pos even).other := by decide

/-- `hi < c` without `lo ≤ hi` does not give strict disfavouring at every share (c = 1, hi = 0,
lo = 5: with no shapers the rule read favours the shapers strictly, six to two). -/
theorem disfavours_needs_step :
    (drawn 1 0 1 0 5 one_pos noShapers).other < (drawn 1 0 1 0 5 one_pos noShapers).carrier := by
  decide

/-- Without `lo ≤ hi`, a start under the rule read there that favours the shapers strictly is not
locked at step 1, and the rule in force at step 2 is below the start's (c = 1, hi = 0, lo = 3,
one to one, read at every step): the summary paragraph's "one loop" needs `lo ≤ hi`. -/
theorem locks_needs_step :
    (drawn 1 0 1 0 3 one_pos even).other < (drawn 1 0 1 0 3 one_pos even).carrier ∧
    ¬ Locked CarrierCounts.order CarrierCounts.favours (drawnBack 1 0 1 0 3 one_pos)
        (drawnCourse 1 0 1 0 3 one_pos (fun _ => true) even 1) ∧
    ¬ CarrierCounts.order.le (drawn 1 0 1 0 3 one_pos even)
        (drawnCourse 1 0 1 0 3 one_pos (fun _ => true) even 2).2 :=
  ⟨by decide,
   fun lock =>
    (show ¬ ((drawnCourse 1 0 1 0 3 one_pos (fun _ => true) even 1).2.carrier *
          (drawn 1 0 1 0 3 one_pos (drawnCourse 1 0 1 0 3 one_pos (fun _ => true) even 1).1).other ≤
        (drawn 1 0 1 0 3 one_pos (drawnCourse 1 0 1 0 3 one_pos (fun _ => true) even 1).1).carrier *
          (drawnCourse 1 0 1 0 3 one_pos (fun _ => true) even 1).2.other) by decide) lock.2,
   fun le =>
    (show ¬ ((drawn 1 0 1 0 3 one_pos even).carrier *
          (drawnCourse 1 0 1 0 3 one_pos (fun _ => true) even 2).2.other ≤
        (drawnCourse 1 0 1 0 3 one_pos (fun _ => true) even 2).2.carrier *
          (drawn 1 0 1 0 3 one_pos even).other) by decide) le⟩

/-! ## 2. The exact regimes, without `lo ≤ hi` -/

theorem drawn_lt_iff {a p c hi lo : Nat} {pos : 0 < a} {x : CarrierCounts} :
    (drawn a p c hi lo pos x).carrier < (drawn a p c hi lo pos x).other ↔
      x.carrier * hi + x.other * lo < x.carrier * c + x.other * c := by
  show x.carrier * (a + hi + p) + x.other * (a + lo + p) <
      x.carrier * (a + c + p) + x.other * (a + c + p) ↔ _
  rw [drawn_split x.carrier x.other a p hi lo, drawn_split x.carrier x.other a p c c]
  exact ⟨Nat.lt_of_add_lt_add_right, fun lt => Nat.add_lt_add_right lt _⟩

/-- **Favoured at every share, exactly**: `c ≤ lo` and `c ≤ hi`, with no hypothesis. -/
theorem favours_everywhere_iff' {a p c hi lo : Nat} {pos : 0 < a} :
    (∀ x, CarrierCounts.favours (drawn a p c hi lo pos x)) ↔ c ≤ lo ∧ c ≤ hi :=
  ⟨fun every =>
    have at_none : 0 * c + 1 * c ≤ 0 * hi + 1 * lo := drawn_favours_iff.mp (every noShapers)
    have low : c ≤ lo := by
      rw [Nat.zero_mul, Nat.zero_mul, Nat.one_mul, Nat.one_mul, Nat.zero_add, Nat.zero_add] at at_none
      exact at_none
    have many : (lo + 1) * c + 1 * c ≤ (lo + 1) * hi + 1 * lo :=
      drawn_favours_iff.mp (every ⟨lo + 1, 1, Nat.zero_lt_one⟩)
    ⟨low, match Nat.lt_or_ge hi c with
      | .inl lt =>
        absurd
          (calc (lo + 1) * hi + (lo + 1)
              = (lo + 1) * (hi + 1) := (Nat.mul_succ (lo + 1) hi).symm
            _ ≤ (lo + 1) * c := Nat.mul_le_mul_left _ lt
            _ ≤ (lo + 1) * c + 1 * c := Nat.le_add_right _ _
            _ ≤ (lo + 1) * hi + 1 * lo := many
            _ = (lo + 1) * hi + lo := congrArg _ (Nat.one_mul lo))
          (Nat.not_le_of_gt (Nat.add_lt_add_left (Nat.lt_succ_self lo) _))
      | .inr ge => ge⟩,
   fun both _ => drawn_favours_iff.mpr
    (Nat.add_le_add (Nat.mul_le_mul_left _ both.2) (Nat.mul_le_mul_left _ both.1))⟩

/-- **Favoured at no share (the others favoured strictly at every share), exactly**: `lo < c` and
`hi ≤ c`, with no hypothesis. The boundary `hi = c` (with `lo < c`) belongs to "nowhere", which
`disfavours_everywhere` (`hi < c`) leaves out. -/
theorem strictly_disfavoured_everywhere_iff {a p c hi lo : Nat} {pos : 0 < a} :
    (∀ x, (drawn a p c hi lo pos x).carrier < (drawn a p c hi lo pos x).other) ↔ lo < c ∧ hi ≤ c :=
  ⟨fun every =>
    have at_none : 0 * hi + 1 * lo < 0 * c + 1 * c := drawn_lt_iff.mp (every noShapers)
    have low : lo < c := by
      rw [Nat.zero_mul, Nat.zero_mul, Nat.one_mul, Nat.one_mul, Nat.zero_add, Nat.zero_add] at at_none
      exact at_none
    have many : c * hi + 1 * lo < c * c + 1 * c := drawn_lt_iff.mp (every ⟨c, 1, Nat.zero_lt_one⟩)
    ⟨low, match Nat.lt_or_ge c hi with
      | .inl lt =>
        absurd
          (calc c * c + 1 * c
              = c * c + c := congrArg _ (Nat.one_mul c)
            _ = c * (c + 1) := (Nat.mul_succ c c).symm
            _ ≤ c * hi := Nat.mul_le_mul_left _ lt
            _ ≤ c * hi + 1 * lo := Nat.le_add_right _ _)
          (Nat.not_le_of_gt many)
      | .inr ge => ge⟩,
   fun both x => drawn_lt_iff.mpr
    (Nat.add_lt_add_of_le_of_lt (Nat.mul_le_mul_left _ both.2)
      (Nat.mul_lt_mul_of_pos_left both.1 x.other_pos))⟩

/-- Witness of the boundary: `c = hi = 3`, `lo = 0`: favoured at no share. -/
theorem at_hi_nowhere (x : CarrierCounts) :
    (drawn 1 0 3 3 0 one_pos x).carrier < (drawn 1 0 3 3 0 one_pos x).other :=
  strictly_disfavoured_everywhere_iff.mpr ⟨by decide, Nat.le_refl 3⟩ x

/-! ## 3. `lo > hi`: the rule read reads against the share -/

theorem drawn_against {a p c hi lo : Nat} {pos : 0 < a} (readers : hi ≤ lo) :
    ReadsShare CarrierCounts.order (reverse CarrierCounts.order) (drawnBack a p c hi lo pos) :=
  ClosedCapture.Drawn.pool_against (u := RuleShaping.shaped a hi p c pos)
    (v := RuleShaping.shaped a lo p c pos) (shaped_le_iff.mpr readers)

/-! ## 4. Either way: every course from the rule read at its start is locked one way -/

theorem drawn_locked_either {a p c hi lo : Nat} {pos : 0 < a} (step : lo ≤ hi) {reads : Nat → Bool}
    (x : CarrierCounts) (k : Nat) :
    Locked CarrierCounts.order CarrierCounts.favours (drawnBack a p c hi lo pos)
        (drawnCourse a p c hi lo pos reads x k) ∨
      Locked (reverse CarrierCounts.order) CarrierCounts.disfavours (drawnBack a p c hi lo pos)
        (drawnCourse a p c hi lo pos reads x k) :=
  match Nat.le_total (drawn a p c hi lo pos x).other (drawn a p c hi lo pos x).carrier with
  | .inl fav => .inl (drawn_locks (reads := reads) step fav k).1
  | .inr dis => .inr (drawn_from_below (reads := reads) step dis k).1

/-! ## 5. The one-loop theorems as one general law of `RuleCapture` (board rule 9) -/

theorem read_start_locks {S : Type} {R : Type} {P : Ideals.Preorder S} {M : Ideals.Preorder R}
    {favours : R → Prop} {f : S → R → S} {h : S → R → R} {reads : Nat → Bool}
    (raises : Raises P favours f) (monotone : ReadsShare P M h)
    (upper : ∀ {g g'}, M.le g g' → favours g → favours g') {x : S} {g : R}
    (fav : favours (h x g)) (k : Nat) :
    Locked M favours h (paced f h reads (x, h x g) k) ∧
      P.le x (paced f h reads (x, h x g) k).1 ∧ M.le (h x g) (paced f h reads (x, h x g) k).2 :=
  have start : Locked M favours h (paced f h reads (x, h x g) 0) :=
    ⟨fav, monotone x x g (h x g) (P.refl x)⟩
  Eq.subst (motive := fun n => Locked M favours h (paced f h reads (x, h x g) n) ∧
      P.le x (paced f h reads (x, h x g) n).1 ∧ M.le (h x g) (paced f h reads (x, h x g) n).2)
    (Nat.zero_add k)
    ⟨locked_stays raises monotone upper start k, locked_not_below raises monotone upper start k⟩

/-- `drawn_locks`, as an instance. -/
theorem drawn_locks' {a p c hi lo : Nat} {pos : 0 < a} (step : lo ≤ hi) {reads : Nat → Bool}
    {x : CarrierCounts} (fav : CarrierCounts.favours (drawn a p c hi lo pos x)) (k : Nat) :
    Locked CarrierCounts.order CarrierCounts.favours (drawnBack a p c hi lo pos)
        (drawnCourse a p c hi lo pos reads x k) ∧
      CarrierCounts.order.le x (drawnCourse a p c hi lo pos reads x k).1 ∧
      CarrierCounts.order.le (drawn a p c hi lo pos x) (drawnCourse a p c hi lo pos reads x k).2 :=
  read_start_locks (g := x) CarrierCounts.raises (drawn_reads_iff.mpr step) CarrierCounts.upper fav k

/-- `drawn_from_below`, as an instance. -/
theorem drawn_from_below' {a p c hi lo : Nat} {pos : 0 < a} (step : lo ≤ hi) {reads : Nat → Bool}
    {x : CarrierCounts} (dis : CarrierCounts.disfavours (drawn a p c hi lo pos x)) (k : Nat) :
    Locked (reverse CarrierCounts.order) CarrierCounts.disfavours (drawnBack a p c hi lo pos)
        (drawnCourse a p c hi lo pos reads x k) ∧
      CarrierCounts.order.le (drawnCourse a p c hi lo pos reads x k).1 x ∧
      CarrierCounts.order.le (drawnCourse a p c hi lo pos reads x k).2 (drawn a p c hi lo pos x) :=
  read_start_locks (g := x) CarrierCounts.lowers (drawn_reads_iff.mpr step).reverse
    CarrierCounts.lower dis k

/-- `ChoiceCountCapture.lower_strict` already states `DrawnShaping.lower_strictly`. -/
theorem lower_strictly_is_lower_strict :
    ∀ {g g' : CarrierCounts}, CarrierCounts.order.le g' g → g.carrier < g.other →
      g'.carrier < g'.other :=
  @ChoiceCountCapture.lower_strict

/-! ## 6. Witnesses the readings lean on -/

/-- From one to three the course is locked the other way at every step (reading (b)'s course). -/
theorem few_locks {reads : Nat → Bool} (k : Nat) :
    Locked (reverse CarrierCounts.order) CarrierCounts.disfavours
        (drawnBack 1 0 1 3 0 CriticalMass.one_pos)
        (drawnCourse 1 0 1 3 0 CriticalMass.one_pos reads CriticalMass.oneToThree k) ∧
      CarrierCounts.order.le
        (drawnCourse 1 0 1 3 0 CriticalMass.one_pos reads CriticalMass.oneToThree k).1
        CriticalMass.oneToThree :=
  have locks := drawn_from_below (reads := reads) (Nat.zero_le 3)
    (Nat.le_of_lt CriticalMass.few_disfavours) k
  ⟨locks.1, locks.2.1⟩

/-- At the critical mass both locks hold and the share stays where it is. -/
theorem at_mass_stays {reads : Nat → Bool} (k : Nat) :
    CarrierCounts.order.le CriticalMass.oneToTwo
        (drawnCourse 1 0 1 3 0 CriticalMass.one_pos reads CriticalMass.oneToTwo k).1 ∧
      CarrierCounts.order.le
        (drawnCourse 1 0 1 3 0 CriticalMass.one_pos reads CriticalMass.oneToTwo k).1
        CriticalMass.oneToTwo :=
  ⟨(drawn_locks (reads := reads) (Nat.zero_le 3) (Nat.le_of_eq CriticalMass.at_mass.symm) k).2.1,
   (drawn_from_below (reads := reads) (Nat.zero_le 3) (Nat.le_of_eq CriticalMass.at_mass) k).2.1⟩
