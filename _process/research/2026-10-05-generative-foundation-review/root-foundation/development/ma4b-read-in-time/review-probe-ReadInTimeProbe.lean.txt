/-! ## P1. The step between can still turn the course for good

`a = 1`, nothing shared, a cost of one, `hi = 3`, `lo = 2`, `lo' = 0`, one shaper to three others,
the rule read at every step, the others reading the loss from step 0. -/

def oneToThree : CarrierCounts := ⟨1, 3, Nat.zero_lt_succ 2⟩

/-- Every premise of `one_between`, `kept_early_iff` and `turned_early_iff` holds; step 0 is the
step between (the late reading favours the others strictly at the start and the shapers strictly
at MA4's share of step 1); yet the course read from step 0 is locked the other way at every step
from step 4. -/
theorem between_turns :
    CarrierCounts.favours (DrawnShaping.drawn 1 0 1 3 2 one_pos oneToThree) ∧
    ¬ CarrierCounts.favours (DrawnShaping.drawn 1 0 1 3 0 one_pos
        (DrawnShaping.drawnCourse 1 0 1 3 2 one_pos always oneToThree 0).1) ∧
    ¬ CarrierCounts.disfavours (DrawnShaping.drawn 1 0 1 3 0 one_pos
        (DrawnShaping.drawnCourse 1 0 1 3 2 one_pos always oneToThree 1).1) ∧
    ∀ k, Locked (reverse CarrierCounts.order) CarrierCounts.disfavours
      (DrawnShaping.drawnRead 1 0 1 3 0 one_pos)
      ((ReadInTime.lossCourse 1 0 1 3 2 0 0 one_pos always oneToThree (2 + 1 + (1 + k))).1.1,
       (ReadInTime.lossCourse 1 0 1 3 2 0 0 one_pos always oneToThree (2 + 1 + (1 + k))).2) :=
  ⟨show 8 ≤ 13 by decide, show ¬ 8 ≤ 7 by decide, show ¬ 76 ≤ 74 by decide,
   (ReadInTime.turned_iff (fB := Beyond.lossStep) (a := 1) (p := 0) (c := 1) (hi := 3) (lo := 2)
      (lo' := 0) (N := 0) (pos := one_pos) (reads := always)
      (s := ((oneToThree, 0), DrawnShaping.drawn 1 0 1 3 2 one_pos oneToThree))
      (r := 2) Thinned.lossStep_accrues (Nat.zero_le 3) rfl rfl
      (Nat.zero_le _) (show 556 ≤ 566 by decide)).mpr (show 41872 ≤ 42248 by decide)⟩

/-- The module's own laws place the reading from step 0 neither among those that keep the lock
from step 1 nor among those that turn the course from step 2. -/
theorem between_neither :
    (¬ ∀ k, Locked CarrierCounts.order CarrierCounts.favours (DrawnShaping.drawnRead 1 0 1 3 0 one_pos)
        ((ReadInTime.lossCourse 1 0 1 3 2 0 0 one_pos always oneToThree (0 + (1 + k))).1.1,
         (ReadInTime.lossCourse 1 0 1 3 2 0 0 one_pos always oneToThree (0 + (1 + k))).2)) ∧
    (¬ ∀ k, Locked (reverse CarrierCounts.order) CarrierCounts.disfavours
        (DrawnShaping.drawnRead 1 0 1 3 0 one_pos)
        ((ReadInTime.lossCourse 1 0 1 3 2 0 0 one_pos always oneToThree (0 + 1 + (1 + k))).1.1,
         (ReadInTime.lossCourse 1 0 1 3 2 0 0 one_pos always oneToThree (0 + 1 + (1 + k))).2)) :=
  ⟨fun kept => between_turns.2.1
      ((ReadInTime.kept_early_iff (show 2 ≤ 3 by decide) (Nat.zero_le 3) between_turns.1 rfl).mp kept),
   fun turned => between_turns.2.2.1
      ((ReadInTime.turned_early_iff (show 2 ≤ 3 by decide) (Nat.zero_le 3) between_turns.1 rfl rfl).mp
        turned)⟩

/-- And `one_between` names step 0 as the only step between. -/
theorem only_zero {M : Nat}
    (atM : ¬ CarrierCounts.favours (DrawnShaping.drawn 1 0 1 3 0 one_pos
        (DrawnShaping.drawnCourse 1 0 1 3 2 one_pos always oneToThree M).1) ∧

/-! ## P7. `deadline`: `some` and `strict` are needed; `covers` is not -/

def noShaper : CarrierCounts := ⟨0, 1, Nat.zero_lt_one⟩

/-- No shaper: `a = 1`, a cost of one, `hi = lo = 2`, `lo' = 0`, from none to one; the start's rule
favours the shapers strictly (three to two), the deadline is `2 * 1 * 1 = 2`, and from the reading
at step 2 the rule in force at step 3 favours the others. -/
theorem deadline_needs_some :
    (DrawnShaping.drawn 1 0 1 2 2 one_pos noShaper).other < (DrawnShaping.drawn 1 0 1 2 2 one_pos noShaper).carrier ∧
    (DrawnShaping.drawn 1 0 1 2 2 one_pos noShaper).other * 1 * noShaper.other = 2 ∧
    ¬ Locked CarrierCounts.order CarrierCounts.favours (DrawnShaping.drawnRead 1 0 1 2 0 one_pos)
        ((ReadInTime.lossCourse 1 0 1 2 2 0 2 one_pos always noShaper (2 + (1 + 0))).1.1,
         (ReadInTime.lossCourse 1 0 1 2 2 0 2 one_pos always noShaper (2 + (1 + 0))).2) :=
  ⟨show 2 < 3 by decide, rfl, fun lock => (show ¬ 8 ≤ 4 by decide) lock.1⟩

/-- Read only at step 12. -/
def onlyTwelve : Nat → Bool
  | 12 => true
  | _ => false

/-- Weakly favoured: `a = 1`, a cost of two, `hi = 3`, `lo = 1`, `lo' = 0`, one to one: the start's
rule is six to six, the deadline is `6 * 2 * 1 = 12`, and the others reading the loss at step 12
leave a rule favouring the others at step 13. -/
theorem deadline_needs_strict :
    CarrierCounts.favours (DrawnShaping.drawn 1 0 2 3 1 one_pos oneToOne) ∧
    ¬ (DrawnShaping.drawn 1 0 2 3 1 one_pos oneToOne).other < (DrawnShaping.drawn 1 0 2 3 1 one_pos oneToOne).carrier ∧
    (DrawnShaping.drawn 1 0 2 3 1 one_pos oneToOne).other * 2 * oneToOne.other = 12 ∧ onlyTwelve 12 = true ∧
    ¬ Locked CarrierCounts.order CarrierCounts.favours (DrawnShaping.drawnRead 1 0 2 3 0 one_pos)
        ((ReadInTime.lossCourse 1 0 2 3 1 0 12 one_pos onlyTwelve oneToOne (12 + (1 + 0))).1.1,
         (ReadInTime.lossCourse 1 0 2 3 1 0 12 one_pos onlyTwelve oneToOne (12 + (1 + 0))).2) :=
  ⟨show 6 ≤ 6 by decide, show ¬ 6 < 6 by decide, rfl, rfl,
   fun lock => (show ¬ 13060694016 ≤ 10883911680 by decide) lock.1⟩

variable {a p c hi lo lo' : Nat} {pos : 0 < a}

/-- **`covers` follows from `step` and `strict`.** A start whose rule read favours the shapers
strictly, with `lo ≤ hi`, has `c < hi`. -/
theorem covers_of_strict (step : lo ≤ hi) {x : CarrierCounts}
    (strict : (DrawnShaping.drawn a p c hi lo pos x).other < (DrawnShaping.drawn a p c hi lo pos x).carrier) :
    c < hi :=
  match Nat.lt_or_ge c hi with
  | .inl h => h
  | .inr h =>
    absurd strict (Nat.not_lt_of_le
      (show x.carrier * (a + hi + p) + x.other * (a + lo + p) ≤ x.carrier * (a + c + p) + x.other * (a + c + p) from
        Nat.add_le_add
          (Nat.mul_le_mul_left _ (Nat.add_le_add_right (Nat.add_le_add_left h a) p))
          (Nat.mul_le_mul_left _ (Nat.add_le_add_right (Nat.add_le_add_left (Nat.le_trans step h) a) p))))

/-- So `deadline` holds without `covers`. -/
theorem deadline' {reads : Nat → Bool} (step : lo ≤ hi) (readers : lo' ≤ hi) {x : CarrierCounts}
    (some : 0 < x.carrier)
    (strict : (DrawnShaping.drawn a p c hi lo pos x).other < (DrawnShaping.drawn a p c hi lo pos x).carrier)
    {N : Nat} (late : (DrawnShaping.drawn a p c hi lo pos x).other * c * x.other ≤ N)
    (at_N : reads N = true) (k : Nat) :
    Locked CarrierCounts.order CarrierCounts.favours (DrawnShaping.drawnRead a p c hi lo' pos)
      ((ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + (1 + k))).1.1,
        (ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + (1 + k))).2) :=
  ReadInTime.deadline step readers (covers_of_strict step strict) some strict late at_N k

/-! ## P8. Premises `one_between` and `closed_for_every` do not need -/

/-- A sequence of rules that never falls has at most one step at which it favours the others
strictly and the shapers strictly at the next. -/
theorem at_most_one_up {g : Nat → CarrierCounts} (up : ∀ t d, CarrierCounts.order.le (g t) (g (t + d)))
    {N M : Nat} (atN : ¬ CarrierCounts.favours (g N) ∧ ¬ CarrierCounts.disfavours (g (N + 1)))
    (atM : ¬ CarrierCounts.favours (g M) ∧ ¬ CarrierCounts.disfavours (g (M + 1))) : N = M :=
  have after : ∀ {n m : Nat}, ¬ CarrierCounts.disfavours (g (n + 1)) → n + 1 ≤ m → CarrierCounts.favours (g m) :=
    fun {n m} strict le =>
      CarrierCounts.upper
        (Eq.subst (motive := fun j => CarrierCounts.order.le (g (n + 1)) (g j)) (add_sub_of_le le)
          (up (n + 1) (m - (n + 1))))
        (Nat.le_of_lt (Nat.gt_of_not_le strict))
  Nat.le_antisymm (Nat.le_of_not_lt fun lt => atN.1 (after atM.2 lt))
    (Nat.le_of_not_lt fun lt => atM.1 (after atN.2 lt))

/-- A sequence of rules that never rises has no such step. -/
theorem none_down {g : Nat → CarrierCounts} (down : ∀ t, CarrierCounts.order.le (g (t + 1)) (g t)) {N : Nat}
    (atN : ¬ CarrierCounts.favours (g N) ∧ ¬ CarrierCounts.disfavours (g (N + 1))) : False :=
  atN.1 (CarrierCounts.upper (down N) (Nat.le_of_lt (Nat.gt_of_not_le atN.2)))

/-- **`one_between` with `lo ≤ hi` alone**: neither `lo' ≤ hi` nor the start favoured is needed.
MA4's course is monotone one way or the other, and the late reading reads the share one way or the
other, so the late reading along the course never falls or never rises. -/
theorem one_between' {reads : Nat → Bool} (step : lo ≤ hi) {x : CarrierCounts} {N M : Nat}
    (atN : ¬ CarrierCounts.favours
        (DrawnShaping.drawn a p c hi lo' pos (DrawnShaping.drawnCourse a p c hi lo pos reads x N).1) ∧
      ¬ CarrierCounts.disfavours
        (DrawnShaping.drawn a p c hi lo' pos (DrawnShaping.drawnCourse a p c hi lo pos reads x (N + 1)).1))
    (atM : ¬ CarrierCounts.favours
        (DrawnShaping.drawn a p c hi lo' pos (DrawnShaping.drawnCourse a p c hi lo pos reads x M).1) ∧
      ¬ CarrierCounts.disfavours
        (DrawnShaping.drawn a p c hi lo' pos (DrawnShaping.drawnCourse a p c hi lo pos reads x (M + 1)).1)) :
    N = M :=
  have mono := (DrawnShaping.drawn_reads_iff (a := a) (p := p) (c := c) (pos := pos)).mpr step
  match Nat.le_total (DrawnShaping.drawn a p c hi lo pos x).other (DrawnShaping.drawn a p c hi lo pos x).carrier,
    Nat.le_total lo' hi with
  | .inl fav, .inl readers => ReadInTime.one_between step readers fav atN atM
  | .inl fav, .inr against =>
    have up : ∀ t d, CarrierCounts.order.le (DrawnShaping.drawnCourse a p c hi lo pos reads x t).1
        (DrawnShaping.drawnCourse a p c hi lo pos reads x (t + d)).1 := fun t d =>
      (locked_not_below CarrierCounts.raises mono CarrierCounts.upper
        (DrawnShaping.drawn_locks (reads := reads) step fav t).1 d).1
    absurd (none_down (g := fun t => DrawnShaping.drawn a p c hi lo' pos
        (DrawnShaping.drawnCourse a p c hi lo pos reads x t).1)
      (fun t => Drawn.pool_against (DrawnShaping.shaped_le_iff.mpr against) _ _
        (DrawnShaping.drawn a p c hi lo pos x) (DrawnShaping.drawn a p c hi lo pos x) (up t 1)) atN) id
  | .inr dis, .inl readers =>
    have down : ∀ t d, CarrierCounts.order.le (DrawnShaping.drawnCourse a p c hi lo pos reads x (t + d)).1
        (DrawnShaping.drawnCourse a p c hi lo pos reads x t).1 := fun t d =>
      (locked_not_below CarrierCounts.lowers mono.reverse CarrierCounts.lower
        (DrawnShaping.drawn_from_below (reads := reads) step dis t).1 d).1
    absurd (none_down (g := fun t => DrawnShaping.drawn a p c hi lo' pos
        (DrawnShaping.drawnCourse a p c hi lo pos reads x t).1)
      (fun t => Drawn.pool_reads (DrawnShaping.shaped_le_iff.mpr readers) _ _
        (DrawnShaping.drawn a p c hi lo pos x) (DrawnShaping.drawn a p c hi lo pos x) (down t 1)) atN) id
  | .inr dis, .inr against =>
    have down : ∀ t d, CarrierCounts.order.le (DrawnShaping.drawnCourse a p c hi lo pos reads x (t + d)).1
        (DrawnShaping.drawnCourse a p c hi lo pos reads x t).1 := fun t d =>
      (locked_not_below CarrierCounts.lowers mono.reverse CarrierCounts.lower
        (DrawnShaping.drawn_from_below (reads := reads) step dis t).1 d).1
    at_most_one_up (g := fun t => DrawnShaping.drawn a p c hi lo' pos
        (DrawnShaping.drawnCourse a p c hi lo pos reads x t).1)
      (fun t d => Drawn.pool_against (DrawnShaping.shaped_le_iff.mpr against) _ _
        (DrawnShaping.drawn a p c hi lo pos x) (DrawnShaping.drawn a p c hi lo pos x) (down t d)) atN atM

/-- **`closed_for_every` without the start favoured**: with `lo ≤ hi`, a step at which the least late
reading favours the shapers forces the start's rule read to favour them. -/
theorem closed_for_every' {reads : Nat → Bool} (step : lo ≤ hi) (readers : lo' ≤ hi) {x : CarrierCounts} {s : Nat}
    (closed : CarrierCounts.favours
      (DrawnShaping.drawn a p c hi 0 pos (DrawnShaping.drawnCourse a p c hi lo pos reads x s).1))
    (d : Nat) (at_N : reads (s + d) = true) (k : Nat) :
    Locked CarrierCounts.order CarrierCounts.favours (DrawnShaping.drawnRead a p c hi lo' pos)
      ((ReadInTime.lossCourse a p c hi lo lo' (s + d) pos reads x (s + d + (1 + k))).1.1,
        (ReadInTime.lossCourse a p c hi lo lo' (s + d) pos reads x (s + d + (1 + k))).2) :=
  have fav : CarrierCounts.favours (DrawnShaping.drawn a p c hi lo pos x) :=
    match Nat.le_total (DrawnShaping.drawn a p c hi lo pos x).other (DrawnShaping.drawn a p c hi lo pos x).carrier with
    | .inl f => f
    | .inr dis => CarrierCounts.upper
        (Drawn.pool_reads (DrawnShaping.shaped_le_iff.mpr step) _ _
          (DrawnShaping.drawn a p c hi lo pos x) (DrawnShaping.drawn a p c hi lo pos x)
          (DrawnShaping.drawn_from_below (reads := reads) step dis s).2.1)
        (ReadInTime.least closed lo)
  ReadInTime.closed_for_every step readers fav closed d at_N k

/-! ## P14 (optional Lean for the fix). The step between starts a new round

Where the late reading favours the shapers at MA4's share of step `N + 1` (as at the step between),
the course read from `N` is decided at step `N + 2` by `kept_iff`: locked in the shapers' favour
from `N + 3` exactly where the late reading favours them at the share of step `N + 2`. Neither the
start favoured nor `lo ≤ hi` is needed. -/

theorem between_round {reads : Nat → Bool} (readers : lo' ≤ hi) {x : CarrierCounts} {N : Nat}
    (late1 : CarrierCounts.favours
        (DrawnShaping.drawn a p c hi lo' pos (DrawnShaping.drawnCourse a p c hi lo pos reads x (N + 1)).1))
    (at_next : reads (N + 1) = true) (at_two : reads (N + 2) = true) :
    (∀ k, Locked CarrierCounts.order CarrierCounts.favours (DrawnShaping.drawnRead a p c hi lo' pos)
        ((ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + 2 + (1 + k))).1.1,
          (ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + 2 + (1 + k))).2)) ↔
      CarrierCounts.favours (DrawnShaping.drawn a p c hi lo' pos
        (ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + 2)).1.1) :=
  have grows := fun t => ClosedCapture.loss_grows (fA := CarrierCounts.select) (fB := Beyond.lossStep)
    (h := ReadInTime.lossRead a p c hi lo lo' N pos) (reads := reads) Thinned.lossStep_accrues
    ((x, 0), DrawnShaping.drawn a p c hi lo pos x) t
  have past1 : N ≤ (ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + 1)).1.2 :=
    Nat.le_trans (Nat.le_trans (Nat.le_succ N) (Nat.le_of_eq (Nat.zero_add (N + 1)).symm)) (grows (N + 1))
  have past2 : N ≤ (ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + 2)).1.2 :=
    Nat.le_trans (Nat.le_trans (Nat.le_add_right N 2) (Nat.le_of_eq (Nat.zero_add (N + 2)).symm)) (grows (N + 2))
  have rule2 : (ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + 1 + 1)).2 =
      Drawn.pool (RuleShaping.shaped a hi p c pos) (RuleShaping.shaped a lo' p c pos)
        (ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + 1)).1.1 :=
    Thinned.rule_read Thinned.lossStep_accrues at_next past1
  have share1 := ReadInTime.next_share (a := a) (p := p) (c := c) (hi := hi) (lo := lo) (lo' := lo')
    (N := N) (pos := pos) (reads := reads) (x := x)
  have fav2 : CarrierCounts.favours (ReadInTime.lossCourse a p c hi lo lo' N pos reads x (N + 2)).2 :=
    Eq.subst (motive := CarrierCounts.favours) rule2.symm
      (Eq.subst (motive := fun y => CarrierCounts.favours (DrawnShaping.drawn a p c hi lo' pos y)) share1.symm late1)
  (ReadInTime.kept_iff Thinned.lossStep_accrues readers at_two past2 fav2).trans
    (DrawnShaping.drawn_favours_iff (a := a) (p := p) (pos := pos)).symm

/-! ## P15 (for the kit, which cites it). In-time steps come first

`study_ma4b.txt`, `texts_ma4b.py` and `make_records_ma4b.py` cite `ReadInTime.in_time_earlier`, which
the module does not contain. This is one statement of it, a corollary of `late_rises`. -/

theorem in_time_earlier {reads : Nat → Bool} (step : lo ≤ hi) (readers : lo' ≤ hi) {x : CarrierCounts}
    (fav : CarrierCounts.favours (DrawnShaping.drawn a p c hi lo pos x)) {s t : Nat} (le : s ≤ t)
    (later : CarrierCounts.disfavours
      (DrawnShaping.drawn a p c hi lo' pos (DrawnShaping.drawnCourse a p c hi lo pos reads x t).1)) :
    CarrierCounts.disfavours
      (DrawnShaping.drawn a p c hi lo' pos (DrawnShaping.drawnCourse a p c hi lo pos reads x s).1) :=
  CarrierCounts.lower
    (Eq.subst (motive := fun j => CarrierCounts.order.le
        (DrawnShaping.drawn a p c hi lo' pos (DrawnShaping.drawnCourse a p c hi lo pos reads x s).1)
        (DrawnShaping.drawn a p c hi lo' pos (DrawnShaping.drawnCourse a p c hi lo pos reads x j).1))
      (add_sub_of_le le) (ReadInTime.late_rises step readers fav s (t - s)))
    later

end ReadInTimeProbe
