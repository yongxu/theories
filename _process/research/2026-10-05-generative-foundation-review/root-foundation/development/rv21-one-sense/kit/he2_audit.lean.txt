-- The kept flame names: on both models the successor is fixed and the alike relation is a class, so the core
-- there is a class the step goes alike across (a 理), by «定后则类之核同行» (not package declarations).
theorem he2_flame_fixed : «定后» Flame.step := by
  intro y z ⟨x, h1, h2⟩
  cases x <;> cases y <;> cases z <;> first | rfl | exact False.elim h1 | exact False.elim h2

theorem he2_twoFlames_fixed : «定后» TwoFlames.step := by
  intro y z ⟨x, ⟨e1, h1⟩, ⟨e2, h2⟩⟩
  exact Prod.ext (e1.symm.trans e2) (he2_flame_fixed y.2 z.2 ⟨x.2, h1, h2⟩)

theorem he2_twoFlames_alike_class : «类» TwoFlames.alike :=
  ⟨fun _ => rfl, fun h => Eq.symm h, fun h k => Eq.trans h k⟩

theorem he2_flame_core_li : «类» («核» Flame.step Flame.alike) ∧ «则» («核» Flame.step Flame.alike) Flame.alike ∧
    «同行» Flame.step («核» Flame.step Flame.alike) :=
  «定后则类之核同行» Flame.alike_class he2_flame_fixed
theorem he2_twoFlames_core_li : «类» («核» TwoFlames.step TwoFlames.alike) ∧
    «则» («核» TwoFlames.step TwoFlames.alike) TwoFlames.alike ∧ «同行» TwoFlames.step («核» TwoFlames.step TwoFlames.alike) :=
  «定后则类之核同行» he2_twoFlames_alike_class he2_twoFlames_fixed
