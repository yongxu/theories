import Lean
import Becoming.Drift
import Becoming.SelectionLevels
open Lean

partial def reach (env : Environment) : List Name → NameSet → NameSet
  | [], s => s
  | n :: rest, s =>
    if s.contains n || !((`Becoming).isPrefixOf n) then reach env rest s else
    let s := s.insert n
    match env.find? n with
    | none => reach env rest s
    | some ci => reach env (ci.getUsedConstantsAsSet.toList ++ rest) s

def modelPrefixes : List Name :=
  [`Becoming.Flow, `Becoming.Web, `Becoming.World, `Becoming.Dependence, `Becoming.WebAt,
   `Becoming.Exchange]

#eval show CoreM Unit from do
  let env ← getEnv
  for row in [`Becoming.SelectionTerm.covariation, `Becoming.SelectionLevels.levels_sized,
      `Becoming.SelectionLevels.rises_exact_iff_levels, `Becoming.Drift.falls_back] do
    let names := (reach env [row] {}).toList
    let model := names.filter fun n => modelPrefixes.any (·.isPrefixOf n)
    let mods := (names.filterMap fun n =>
      (env.getModuleIdxFor? n).map fun i => env.header.moduleNames[i.toNat]!).eraseDups
    IO.println s!"{row}: {names.length} Becoming constants; model constants {model}; modules {mods}"
    let outside := names.filter fun n => match env.getModuleIdxFor? n with
      | some i => let m := env.header.moduleNames[i.toNat]!
                  m != `Becoming.Prelude && m != `Becoming.GrammarLaws
      | none => false
    IO.println s!"  outside the prelude and grammar laws: {outside}"
