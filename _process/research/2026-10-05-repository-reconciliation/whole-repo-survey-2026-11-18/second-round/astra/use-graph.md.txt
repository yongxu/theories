from pathlib import Path
from collections import Counter
from functools import cache
import hashlib, json, re

root = Path("experiments/integrated-correction/general-framework/generative-basis/lean")
r = json.loads((root/"becoming/becoming-verification.json").read_text())
idx = json.load(open("theories/becoming/results/data/lean-index.json"))["contents_by_sha256"]
paths = r["source_paths"]
names, rows, imports, sources, private = {}, {}, {}, {}, set()

def clean(s):
    patterns = [r'/-|--[^\n]*|"(?:\\.|[^"\\])*"', r'/-|-/']
    pos = depth = 0
    out = []
    while h := re.compile(patterns[bool(depth)]).search(s,pos):
        out.append("\n"*s.count("\n",pos,h.start()) if depth else s[pos:h.start()])
        out.append(" "+"\n"*h[0].count("\n"))
        depth += (h[0]=="/-")-(h[0]=="-/")
        pos = h.end()
    assert depth == 0
    return "".join(out)+s[pos:]

def norm(s): return s.replace("«","").replace("»","")

for m,p in paths.items():
    b = (root/p).read_bytes()
    h = hashlib.sha256(b).hexdigest()
    assert h == r["source_sha256"][m]
    j = idx[h]
    sources[m] = clean(b.decode()).splitlines()
    imports[m] = set(j["imports"]) & paths.keys()
    rows[m] = {d["line"]:d for d in j["declarations"]}
    for i,d in rows[m].items():
        q = norm(d["qualified_name_hint"])
        names[q] = (m,i,d["kind"])
        if "private " in sources[m][i-1]: private.add(q)

@cache
def visible(m): return {m}.union(*(visible(p) for p in imports[m]))

def prefixes(s):
    while s:
        yield s+"."
        s = s.rpartition(".")[0]
    yield ""

spaces = {p for q in names for p in prefixes(q.rpartition(".")[0])}
atom = r"(?:«[^»\n]+»|[^\W\d][\w'₀-₉]*)"
tokens = re.compile(r"(?<![\w'.])"+atom+r"(?:\."+atom+r")*")
binders = re.compile(r"[(\{]\s*((?:[\w']+\s+)*[\w']+)\s*:(?!=)")
stop = r"^(?:#|@\[|(?:namespace|section|mutual|end|open|export|universe|variable|omit|include|attribute|set_option|noncomputable|instance|example|local|notation|private|protected)\b)"
edges = set()
for m,lines in sources.items():
    ns = ""; opens = once = (); stack = []; spans = []; active = False
    @cache
    def lookup(t,ns,opens):
        while t:
            for group in [*((p,) for p in prefixes(ns)), opens]:
                hits = {p+t for p in group if p+t in names and names[p+t][0] in visible(m)
                        and (p+t not in private or names[p+t][0]==m)}
                if hits: return next(iter(hits)) if len(hits)==1 else None
            t = t.rpartition(".")[0]
    for i,line in enumerate(lines,1):
        if d := rows[m].get(i):
            q = norm(d["qualified_name_hint"])
            body = [re.split(r"\b"+d["kind"]+r"\s+"+re.escape(d["name"]),line,maxsplit=1)[1]]

            active = True; once = ()
        elif re.match(stop,line):

            w = line.split()
            if w[0] in ("namespace","section","mutual"):
                stack.append((ns,opens))
                if w[0]=="namespace": ns=(ns+"."+w[1]).strip(".")
            elif w[0]=="end" and stack: ns,opens=stack.pop()
            elif re.fullmatch(r"open\s+[\w. ]+",line):
                temp = w[-1]=="in"
                ps = tuple(next((p+n+"." for p in prefixes(ns) if p+n+"." in spaces),"@")
                           for n in w[1:len(w)-temp])
                if temp: once+=ps
                else: opens+=ps
        elif active: body.append(line)
    for q,ns,op,body in spans:
        s = "\n".join(body)
        local = {n for g in binders.findall(s) for n in g.split()}
        for t in set(tokens.findall(s)):
            if t.split(".")[0] in local: continue
            v = lookup(norm(t).removeprefix("_root_."),ns,op)
            if v and (names[v][0]!=m or names[v][1]<names[q][1]): edges.add((q,v))

th = {q for q in names if names[q][2] in ("theorem","lemma")}
for es in (edges, {(q,v) for q,v in edges if q in th and v in th}):
    uses = Counter(v for q,v in es)

pairs = Counter((names[q][0],names[v][0]) for q,v in edges if names[q][0]!=names[v][0])
for (caller,target),count in sorted(pairs.items()): print(caller,target,count)
