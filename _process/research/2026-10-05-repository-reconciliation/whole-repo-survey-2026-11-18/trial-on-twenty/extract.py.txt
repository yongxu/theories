# Extracts, for the trial on twenty, each theorem's signature, docstring, record law and chapter paragraph. AI-written (Claude Fable).
import json, re, sys
ROOT='/Users/ren/theories/'; LEAN=ROOT+'experiments/integrated-correction/general-framework/generative-basis/lean/'
R=ROOT+'tmp/repo-picture-2026-11-18/round2/'
def load(p):
    s=open(p).read(); return json.loads(s[s.index('{'):s.rindex('}')+1]) if not s.lstrip().startswith('[') else json.loads(s)
sample={x['id']:x for x in load(R+'sample-H.json')}
C={x['id']:x for x in load(R+'claude/tellings-H.json')['items']}
A={x['id']:x for x in load(R+'astra/tellings-H.json')['items']}
out=[]; problems=[]
for i,(rid,c) in enumerate(C.items()):
    a=A.get(rid,{})
    flagged=sorted({t for t in ('record','chapter','docstring') if c.get(t)=='says-more' or a.get(t)=='says-more'})
    if not flagged: continue
    m=re.match(r'(\S+) \(([^:()]+\.lean):(\d+)\)',c['declaration'])
    if not m: problems.append((rid,'declaration unparsed',c['declaration'])); continue
    name,path,line=m.group(1),m.group(2),int(m.group(3))
    try: L=open(LEAN+path).read().split('\n')
    except Exception as e: problems.append((rid,'lean file',path)); continue
    k=line-1
    short=name.split('.')[-1]
    if short not in L[k]:
        hits=[j for j in range(max(0,k-5),min(len(L),k+6)) if re.search(r'\b(theorem|lemma|def)\b.*'+re.escape(short),L[j])]
        if hits: k=hits[0]
        else: problems.append((rid,'name not at line',f'{path}:{line}'))
    sig=[]; j=k
    while j<len(L) and j<k+45:
        sig.append(L[j])
        if re.search(r':=\s*(by)?\s*$',L[j]) or ' := ' in L[j] or L[j].rstrip().endswith(' where') or L[j].strip().startswith('|'): break
        j+=1
    doc=[]; j=k-1
    while j>=0 and L[j].strip()=='' : j-=1
    if j>=0 and L[j].rstrip().endswith('-/'):
        e=j
        while j>=0 and '/--' not in L[j] and '/-' not in L[j]: j-=1
        doc=L[j:e+1]
    rec=json.load(open(ROOT+sample[rid]['record']))
    law=None
    for lw in rec.get('laws',[]):
        if any(short in d or name in d for d in lw.get('declarations',[])): law=lw.get('en'); break
    cf,cl=sample[rid]['chapter_home'].split(':'); CH=open(ROOT+cf).read().split('\n'); q=int(cl)-1
    while q<len(CH) and (CH[q].strip()=='' or CH[q].lstrip().startswith('<a id')): q+=1
    para=CH[q]
    out.append({'n':len(out)+1,'id':rid,'declaration':name,'lean_file':path,'line':k+1,'hypotheses_fable':c['statement_gist'].split(':')[0],
      'signature':'\n'.join(sig),'docstring':'\n'.join(doc),'record_statement':rec.get('statement',{}).get('en',''),'record_law':law or '',
      'chapter_file':cf,'chapter_line':int(cl),'chapter_paragraph':para,'flagged':flagged,
      'faults':{t:{'fable':c.get(t+'_fault',''),'astra':a.get(t+'_fault','')} for t in flagged},
      'verdicts':{t:{'fable':c.get(t),'astra':a.get(t)} for t in ('record','chapter','docstring')}})
json.dump(out,open(ROOT+'tmp/repo-picture-2026-11-18/trial/inputs.json','w'),ensure_ascii=False,indent=1)
print(len(out),'theorems;',sum(len(x['flagged']) for x in out),'tellings flagged; problems:',problems)
for x in out:
    w=lambda s: len(s.split())
    print(x['n'],x['id'][:38].ljust(38),x['declaration'].split('.')[-1][:28].ljust(28),'sig',w(x['signature']),'doc',w(x['docstring']),'law',w(x['record_law']),'ch',w(x['chapter_paragraph']),x['flagged'],x['hypotheses_fable'][:14])
